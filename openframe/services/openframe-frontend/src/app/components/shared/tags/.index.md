Public API barrel file for the tags feature module, re-exporting all components, hooks, utilities, and types related to tag management.

## Key Components

| Export | Source | Description |
|--------|--------|-------------|
| `TagRow` | `tag-row` | Component for rendering a single tag row |
| `TagValueAutocomplete` | `tag-value-autocomplete` | Autocomplete input for tag values |
| `TagsEditor` | `tags-editor` | Full tags editing UI component |
| `useTagKeySuggestions` | `use-tag-key-suggestions` | Hook for fetching tag key suggestions |
| `useTagValueSuggestions` | `use-tag-value-suggestions` | Hook for fetching tag value suggestions |
| `useCreateTagMutation` | `use-tag-mutations` | Hook for creating tags |
| `useUpdateTagMutation` | `use-tag-mutations` | Hook for updating tags |
| `useDeleteTagMutation` | `use-tag-mutations` | Hook for deleting tags |
| `isValidTag`, `validateTagKey`, `validateTagValues` | `tag-row` | Tag validation utilities |
| `TAG_MAX_LENGTH`, `TAG_REGEX` | `tag-row` | Validation constants |
| `Tag`, `TagEntityType`, `TagEntry`, `TagEntryWithId` | `types` | TypeScript types |

## Usage Example

```typescript
import {
  TagsEditor,
  useCreateTagMutation,
  useDeleteTagMutation,
  useTagKeySuggestions,
  validateTagKey,
} from '@/features/tags';
import type { Tag, TagEntityType } from '@/features/tags';

function DeviceTagsPanel({ entityId, entityType }: { entityId: string; entityType: TagEntityType }) {
  const { data: keySuggestions } = useTagKeySuggestions(entityType);
  const createTag = useCreateTagMutation();
  const deleteTag = useDeleteTagMutation();

  return (
    <TagsEditor
      entityId={entityId}
      entityType={entityType}
      keySuggestions={keySuggestions}
      onCreate={(tag: Tag) => createTag.mutate(tag)}
      onDelete={(tagId: string) => deleteTag.mutate(tagId)}
    />
  );
}
```
