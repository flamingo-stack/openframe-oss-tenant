Barrel export file for the Tags feature module, re-exporting all public API components, hooks, utilities, and types related to tag management.

## Key Components

| Export | Source | Description |
|--------|--------|-------------|
| `TagRow` | `tag-row` | UI component for rendering a single tag row |
| `TagValueAutocomplete` | `tag-value-autocomplete` | Autocomplete input for tag values |
| `TagsEditor` | `tags-editor` | Full tag editor UI component |
| `Tag`, `TagEntityType`, `TagEntry`, `TagEntryWithId` | `types` | TypeScript type definitions |
| `useTagKeySuggestions` | `use-tag-key-suggestions` | Hook for fetching tag key suggestions |
| `useTagValueSuggestions` | `use-tag-value-suggestions` | Hook for fetching tag value suggestions |
| `useCreateTagMutation`, `useUpdateTagMutation`, `useDeleteTagMutation` | `use-tag-mutations` | CRUD mutation hooks for tags |
| `isValidTag`, `validateTagKey`, `validateTagValues`, `TAG_MAX_LENGTH`, `TAG_REGEX` | `tag-row` | Validation utilities and constants |

## Usage Example

```typescript
import {
  TagsEditor,
  useCreateTagMutation,
  useTagKeySuggestions,
  isValidTag,
} from '@/features/tags';
import type { Tag, TagEntityType } from '@/features/tags';

const entityType: TagEntityType = 'device';

function DeviceTags({ entityId }: { entityId: string }) {
  const { suggestions } = useTagKeySuggestions(entityType);
  const { mutate: createTag } = useCreateTagMutation();

  return (
    <TagsEditor
      entityId={entityId}
      entityType={entityType}
      keySuggestions={suggestions}
      onAdd={(tag: Tag) => isValidTag(tag) && createTag(tag)}
    />
  );
}
```
