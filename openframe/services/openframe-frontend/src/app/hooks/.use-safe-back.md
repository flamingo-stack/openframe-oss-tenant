Provides safe browser back-navigation utilities for Next.js, using the Navigation API's `canGoBack` to avoid unsafe `router.back()` calls (e.g., fresh tabs, cross-origin history entries, direct URL pastes).

## Key Components

| Export | Type | Description |
|--------|------|-------------|
| `navCanGoBack` | `function` | Probes `window.navigation.canGoBack` defensively; returns `false` for unsupported browsers or when back navigation is unsafe |
| `useSafeBack` | React hook | Returns a callback that calls `router.back()` when safe, otherwise navigates to a `fallback` route via `router.push()` |
| `safeBackOrReplace` | `function` | Post-submit helper that calls `router.back()` when safe, otherwise uses `router.replace(fallback)` to keep the form out of history |

## Usage Example

```typescript
// useSafeBack — general back button
'use client';
import { useSafeBack } from '@/hooks/use-safe-back';

export function BackButton() {
  const handleBack = useSafeBack('/dashboard');
  return <button onClick={handleBack}>Go Back</button>;
}

// safeBackOrReplace — after form submission
'use client';
import { useRouter } from 'next/navigation';
import { safeBackOrReplace } from '@/hooks/use-safe-back';

export function EditForm() {
  const router = useRouter();

  async function onSubmit(data: FormData) {
    await saveChanges(data);
    safeBackOrReplace(router, '/settings');
    // If safe: pops form from history via back()
    // If unsafe: replaces current entry, keeping form out of stack
  }
}
```

> **Browser support note:** `window.navigation.canGoBack` requires Chrome/Edge 102+ or Safari 16.4+. Both `useSafeBack` and `safeBackOrReplace` fall back gracefully on unsupported browsers.
