A React hook for fetching and managing the current release version of the application using TanStack Query with error handling and toast notifications.

## Key Components

- **`useReleaseVersion`** - Main hook that fetches and returns release version data with loading states
- **`fetchReleaseVersion`** - API client function that retrieves version from `/api/release-version` endpoint
- **`releaseVersionQueryKeys`** - Query key factory for consistent cache management
- **`ReleaseVersionResponse`** - Type definition for API response structure
- **`UseReleaseVersionOptions`** - Configuration interface with optional `enabled` flag

## Usage Example

```typescript
import { useReleaseVersion } from '@/hooks/use-release-version'

function AppVersionDisplay() {
  const { 
    releaseVersion, 
    isLoading, 
    error, 
    refetch 
  } = useReleaseVersion()

  if (isLoading) return <div>Loading version...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      <span>Version: {releaseVersion}</span>
      <button onClick={() => refetch()}>
        Refresh Version
      </button>
    </div>
  )
}

// Conditionally enable the query
function ConditionalVersionCheck() {
  const [shouldFetch, setShouldFetch] = useState(false)
  
  const { releaseVersion } = useReleaseVersion({ 
    enabled: shouldFetch 
  })

  return (
    <div>
      <button onClick={() => setShouldFetch(true)}>
        Load Version
      </button>
      {shouldFetch && <span>{releaseVersion}</span>}
    </div>
  )
}
```
