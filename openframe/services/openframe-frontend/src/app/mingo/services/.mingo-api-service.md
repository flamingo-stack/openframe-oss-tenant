API service class that provides React Query mutations for Mingo AI chat operations including dialog management, message sending, and approval request handling.

## Key Components

- **MingoApiService**: Static class providing mutation factory methods for chat operations
- **createDialogMutation()**: Creates new admin chat dialogs
- **sendMessageMutation()**: Sends messages to existing dialogs  
- **approveRequestMutation()**: Approves pending approval requests with toast notifications
- **rejectRequestMutation()**: Rejects pending approval requests with toast notifications
- **TypeScript interfaces**: Defines request/response types for all API operations

## Usage Example

```typescript
import { MingoApiService } from './mingo-api-service'

function ChatComponent() {
  const createDialog = MingoApiService.createDialogMutation()
  const sendMessage = MingoApiService.sendMessageMutation()
  const approveRequest = MingoApiService.approveRequestMutation()

  const handleCreateDialog = async () => {
    try {
      const dialog = await createDialog.mutateAsync()
      console.log('Created dialog:', dialog.id)
    } catch (error) {
      console.error('Failed to create dialog:', error)
    }
  }

  const handleSendMessage = async (dialogId: string, content: string) => {
    await sendMessage.mutateAsync({ dialogId, content })
  }

  const handleApproval = async (requestId: string) => {
    await approveRequest.mutateAsync(requestId)
    // Toast notification handled automatically
  }

  return (
    <div>
      <button onClick={handleCreateDialog} disabled={createDialog.isPending}>
        Create Dialog
      </button>
      {/* Other UI components */}
    </div>
  )
}
```
