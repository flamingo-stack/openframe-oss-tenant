API service class that provides React Query mutations for Mingo AI chat operations, including dialog management, message sending, and approval workflows.

## Key Components

- **MingoApiService** - Static class with mutation factory methods
- **createDialogMutation()** - Creates new admin chat dialogs
- **sendMessageMutation()** - Sends messages to existing dialogs
- **approveRequestMutation()** - Approves pending requests with toast notifications
- **rejectRequestMutation()** - Rejects pending requests with toast notifications

**Type Definitions:**
- `CreateDialogResponse` - Dialog creation response with metadata
- `SendMessageResponse` - Message sending response
- `ApprovalResponse` - Approval operation response
- `CreateDialogRequest` - Dialog creation payload
- `SendMessageRequest` - Message sending payload

## Usage Example

```typescript
import { MingoApiService } from './mingo-api-service'

function ChatComponent() {
  // Create dialog mutation
  const createDialog = MingoApiService.createDialogMutation()
  
  // Send message mutation
  const sendMessage = MingoApiService.sendMessageMutation()
  
  // Approval mutations
  const approveRequest = MingoApiService.approveRequestMutation()
  const rejectRequest = MingoApiService.rejectRequestMutation()

  const handleCreateDialog = async () => {
    const dialog = await createDialog.mutateAsync()
    console.log('Created dialog:', dialog.id)
  }

  const handleSendMessage = async (dialogId: string) => {
    await sendMessage.mutateAsync({
      dialogId,
      content: 'Hello Mingo!'
    })
  }

  const handleApproval = async (requestId: string, approve: boolean) => {
    if (approve) {
      await approveRequest.mutateAsync(requestId)
    } else {
      await rejectRequest.mutateAsync(requestId)
    }
  }

  return (
    <div>
      <button onClick={handleCreateDialog}>
        Create Dialog
      </button>
      {/* UI components */}
    </div>
  )
}
```
