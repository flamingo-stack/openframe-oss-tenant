TypeScript type definitions and utilities for handling messages in the Flamingo chat system, providing interfaces for both GraphQL API responses and core message structures used throughout the application.

## Key Components

- **GraphQLMessage**: Backend message format with dialog metadata and owner information
- **CoreMessage**: Frontend message format with role-based structure for chat UI
- **MessageConnection**: GraphQL connection pattern for paginated message queries
- **MessagesResponse**: Complete API response wrapper for message queries
- **MessagePage**: Transformed page structure for frontend consumption
- **Type Guards**: Runtime type checking utilities (`isGraphQLMessage`, `isCoreMessage`)

## Usage Example

```typescript
import { GraphQLMessage, CoreMessage, isGraphQLMessage } from './message.types'

// Working with API responses
const response: MessagesResponse = await fetchMessages(dialogId)
const messages = response.data.messages.edges.map(edge => edge.node)

// Type-safe message handling
function processMessage(message: GraphQLMessage | CoreMessage) {
  if (isGraphQLMessage(message)) {
    // Handle GraphQL message format
    console.log(`Dialog: ${message.dialogId}, Owner: ${message.owner.type}`)
    return transformToCore(message)
  } else {
    // Handle core message format
    console.log(`Role: ${message.role}, Content: ${message.content}`)
    return message
  }
}

// Pagination handling
function loadMoreMessages(page: MessagePage) {
  if (page.pageInfo.hasNextPage) {
    return fetchMessages(dialogId, { after: page.pageInfo.endCursor })
  }
}
```
