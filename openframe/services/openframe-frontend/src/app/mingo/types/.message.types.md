Defines TypeScript interfaces and type guards for handling messages in the OpenFrame chat system, providing structures for both GraphQL API responses and internal message representation.

## Key Components

- **GraphQLMessage**: Complete message structure from the GraphQL API including dialog metadata, owner information, and chat type
- **CoreMessage**: Simplified internal message representation with role-based typing and optional assistant metadata
- **MessageConnection**: GraphQL connection pattern for paginated message results with cursor-based pagination
- **MessagesResponse**: Wrapper for GraphQL query responses containing message connections
- **MessagePage**: Simplified paginated message structure for internal use
- **Type Guards**: `isGraphQLMessage()` and `isCoreMessage()` functions for runtime type checking

## Usage Example

```typescript
import { GraphQLMessage, CoreMessage, isGraphQLMessage } from './message.types'

// Handle API response
const response: MessagesResponse = await fetchMessages(dialogId)
const messages = response.data.messages.edges.map(edge => edge.node)

// Transform GraphQL message to core format
function transformMessage(message: GraphQLMessage): CoreMessage {
  return {
    id: message.id,
    role: message.owner.type === 'USER' ? 'user' : 'assistant',
    content: message.messageData.content,
    timestamp: new Date(message.createdAt),
    assistantType: message.owner.model as AssistantType
  }
}

// Type-safe message processing
function processMessage(message: unknown) {
  if (isGraphQLMessage(message)) {
    return transformMessage(message)
  } else if (isCoreMessage(message)) {
    return message
  }
  throw new Error('Invalid message format')
}
```
