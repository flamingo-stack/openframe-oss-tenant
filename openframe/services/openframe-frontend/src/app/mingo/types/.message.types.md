Defines TypeScript interfaces and type guards for message handling in the chat/dialog system, bridging GraphQL API responses with frontend message components.

## Key Components

- **`GraphQLMessage`** - Interface for messages received from GraphQL API with dialog metadata
- **`CoreMessage`** - Frontend-optimized message interface with role-based content structure
- **`MessageConnection`** - GraphQL connection pattern for paginated message queries
- **`MessagesResponse`** - Complete GraphQL query response wrapper
- **`MessagePage`** - Simplified pagination structure for frontend consumption
- **Type Guards** - `isGraphQLMessage()` and `isCoreMessage()` for runtime type checking

## Usage Example

```typescript
import { Message, isGraphQLMessage, isCoreMessage } from './message.types'
import type { MessageContent } from '@flamingo-stack/openframe-frontend-core'

// Handle mixed message sources
function processMessage(message: unknown): Message | null {
  if (isGraphQLMessage(message)) {
    // Transform GraphQL message to CoreMessage
    return {
      id: message.id,
      role: message.owner.type === 'user' ? 'user' : 'assistant',
      content: message.messageData as MessageContent,
      timestamp: new Date(message.createdAt)
    }
  }
  
  if (isCoreMessage(message)) {
    // Already in correct format
    return message
  }
  
  return null
}

// Paginated message handling
function handleMessagesResponse(response: MessagesResponse): MessagePage {
  return {
    messages: response.data.messages.edges.map(edge => edge.node),
    pageInfo: response.data.messages.pageInfo
  }
}
```
