This file defines TypeScript interfaces and type guards for handling different message formats in the Flamingo chat system, bridging between GraphQL responses and core application message structures.

## Key Components

**Core Interfaces:**
- `GraphQLMessage` - Message format from GraphQL API with dialog metadata
- `CoreMessage` - Standardized internal message format with role-based structure
- `Message` - Type alias for CoreMessage (main message type used throughout app)
- `MessageConnection` - GraphQL connection pattern for paginated message queries
- `MessagesResponse` - API response wrapper for message queries
- `MessagePage` - Simplified page format for message pagination

**Type Guards:**
- `isGraphQLMessage()` - Runtime check for GraphQL message format
- `isCoreMessage()` - Runtime check for core message format

## Usage Example

```typescript
import { Message, isGraphQLMessage, isCoreMessage } from './message.types'

// Handle incoming message from API
function processMessage(rawMessage: unknown): Message | null {
  if (isGraphQLMessage(rawMessage)) {
    // Transform GraphQL message to core format
    return {
      id: rawMessage.id,
      role: rawMessage.owner.type === 'USER' ? 'user' : 'assistant',
      content: rawMessage.messageData.content,
      timestamp: new Date(rawMessage.createdAt)
    }
  }
  
  if (isCoreMessage(rawMessage)) {
    // Already in correct format
    return rawMessage
  }
  
  return null
}

// Working with paginated messages
function handleMessagesResponse(response: MessagesResponse) {
  const messages = response.data.messages.edges.map(edge => edge.node)
  const hasMore = response.data.messages.pageInfo.hasNextPage
  
  return { messages, hasMore }
}
```
