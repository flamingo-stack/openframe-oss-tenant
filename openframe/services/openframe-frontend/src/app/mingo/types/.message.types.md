TypeScript type definitions for message handling in the OpenFrame chat system, providing interfaces for both GraphQL and core message formats with type guards for differentiation.

## Key Components

- **GraphQLMessage**: Server-side message structure with dialog metadata, owner information, and raw message data
- **CoreMessage**: Client-side message format with role-based typing, content, and UI-specific properties
- **MessageConnection**: GraphQL connection pattern for paginated message lists with cursor-based navigation
- **MessagesResponse**: Response wrapper for GraphQL message queries
- **MessagePage**: Simplified page structure for message pagination
- **Type Guards**: `isGraphQLMessage()` and `isCoreMessage()` functions for runtime type checking

## Usage Example

```typescript
import { Message, isGraphQLMessage, isCoreMessage } from './message.types'

// Handle different message formats
function processMessage(message: GraphQLMessage | CoreMessage) {
  if (isGraphQLMessage(message)) {
    // Server-side message processing
    console.log(`Dialog: ${message.dialogId}, Owner: ${message.owner.type}`)
    return transformToCore(message)
  }
  
  if (isCoreMessage(message)) {
    // Client-side message display
    return {
      ...message,
      displayName: message.name || message.role,
      showAvatar: message.role === 'assistant'
    }
  }
}

// Paginated message handling
function handleMessagePage(response: MessagesResponse): MessagePage {
  const connection = response.data.messages
  return {
    messages: connection.edges.map(edge => edge.node),
    pageInfo: connection.pageInfo
  }
}
```
