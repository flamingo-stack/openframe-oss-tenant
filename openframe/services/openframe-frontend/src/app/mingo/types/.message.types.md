Defines TypeScript interfaces and types for message handling in the Flamingo OpenFrame platform, supporting both GraphQL and core message formats.

## Key Components

### Types & Interfaces
- **`GraphQLMessage`** - Server-side message format with dialog context and owner metadata
- **`CoreMessage`** - Client-side message format with role-based content and assistant types
- **`Message`** - Type alias for CoreMessage (primary message type)
- **`MessageConnection`** - GraphQL connection pattern for paginated message results
- **`MessagesResponse`** - API response wrapper for message queries
- **`MessagePage`** - Processed message page with pagination metadata

### Type Guards
- **`isGraphQLMessage()`** - Runtime check for GraphQL message format
- **`isCoreMessage()`** - Runtime check for core message format

## Usage Example

```typescript
import { Message, isGraphQLMessage, isCoreMessage } from './message.types'

// Handle different message formats
function processMessage(message: unknown): Message | null {
  if (isGraphQLMessage(message)) {
    // Transform GraphQL format to core format
    return {
      id: message.id,
      role: message.owner.type === 'USER' ? 'user' : 'assistant',
      content: message.messageData.content,
      timestamp: new Date(message.createdAt)
    }
  }
  
  if (isCoreMessage(message)) {
    return message // Already in correct format
  }
  
  return null // Invalid format
}

// Create a new message
const userMessage: Message = {
  id: 'msg-123',
  role: 'user',
  content: { text: 'Hello, how can I help?' },
  timestamp: new Date()
}
```
