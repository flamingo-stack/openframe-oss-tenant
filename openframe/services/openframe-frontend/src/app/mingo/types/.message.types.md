TypeScript type definitions for message handling in the Flamingo Stack, providing interfaces for GraphQL messages, core messages, and pagination structures used across the chat system.

## Key Components

- **GraphQLMessage**: Backend message structure with dialog metadata and owner information
- **CoreMessage**: Frontend message format with role-based typing and content structure  
- **MessageConnection**: GraphQL pagination wrapper with cursor-based navigation
- **MessagesResponse**: API response structure for message queries
- **MessagePage**: Simplified pagination format for frontend consumption
- **Type Guards**: `isGraphQLMessage()` and `isCoreMessage()` for runtime type checking

## Usage Example

```typescript
import { Message, isGraphQLMessage, MessagePage } from './message.types'

// Working with different message formats
function processMessage(message: unknown) {
  if (isGraphQLMessage(message)) {
    // Handle GraphQL message from backend
    console.log(`Dialog: ${message.dialogId}, Owner: ${message.owner.type}`)
    return message.messageData
  }
  
  if (isCoreMessage(message)) {
    // Handle frontend message format
    console.log(`Role: ${message.role}, Content: ${message.content}`)
    return message.content
  }
}

// Pagination handling
function handleMessagePage(page: MessagePage) {
  const { messages, pageInfo } = page
  
  if (pageInfo.hasNextPage) {
    // Load more messages using endCursor
    loadMessages({ after: pageInfo.endCursor })
  }
  
  return messages.map(msg => msg.messageData)
}
```
