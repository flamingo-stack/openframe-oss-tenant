TypeScript type definitions for message data structures used in the Flamingo chat/ticketing system, providing interfaces for both GraphQL API responses and core application message formats.

## Key Components

- **`GraphQLMessage`** - Server-side message format with dialog metadata and owner information
- **`CoreMessage`** - Client-side message format with role-based chat structure
- **`MessageConnection`** - GraphQL pagination wrapper for message lists
- **`MessagesResponse`** - Complete API response structure for message queries
- **`MessagePage`** - Simplified pagination format for processed messages
- **Type Guards** - `isGraphQLMessage()` and `isCoreMessage()` for runtime type checking

## Usage Example

```typescript
import { Message, GraphQLMessage, isGraphQLMessage } from './message.types'

// Type-safe message handling
function processMessage(message: Message | GraphQLMessage) {
  if (isGraphQLMessage(message)) {
    // Handle server format
    return {
      id: message.id,
      chatType: message.chatType,
      data: message.messageData
    }
  } else {
    // Handle client format
    return {
      id: message.id,
      role: message.role,
      content: message.content,
      timestamp: message.timestamp
    }
  }
}

// Working with paginated responses
function handleMessagesResponse(response: MessagesResponse) {
  const messages = response.data.messages.edges.map(edge => edge.node)
  const hasMore = response.data.messages.pageInfo.hasNextPage
  
  return { messages, hasMore }
}
```
