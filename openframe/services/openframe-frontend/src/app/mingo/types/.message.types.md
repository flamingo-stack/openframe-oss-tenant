TypeScript type definitions for message handling across GraphQL and core chat systems, providing interfaces for different message formats and pagination structures.

## Key Components

### Main Interfaces
- **`GraphQLMessage`** - Server-side message format with dialog metadata, owner info, and raw message data
- **`CoreMessage`** - Client-side message format with role-based content, assistant types, and UI properties
- **`MessageConnection`** - GraphQL connection pattern for paginated message queries
- **`MessagePage`** - Simplified pagination wrapper for processed messages

### Response Types
- **`MessagesResponse`** - GraphQL query response wrapper
- **`Message`** - Type alias for CoreMessage (primary client message type)

### Type Guards
- **`isGraphQLMessage()`** - Runtime type checking for server messages
- **`isCoreMessage()`** - Runtime type checking for client messages

## Usage Example

```typescript
import { Message, isGraphQLMessage, MessagePage } from './message.types'

// Transform GraphQL message to core message
function transformMessage(gqlMessage: GraphQLMessage): Message {
  return {
    id: gqlMessage.id,
    role: gqlMessage.owner.type === 'user' ? 'user' : 'assistant',
    content: gqlMessage.messageData.content,
    timestamp: new Date(gqlMessage.createdAt),
    assistantType: gqlMessage.owner.model as AssistantType
  }
}

// Process mixed message types
function processMessage(message: unknown): Message | null {
  if (isGraphQLMessage(message)) {
    return transformMessage(message)
  } else if (isCoreMessage(message)) {
    return message
  }
  return null
}

// Handle paginated messages
function renderMessagePage(page: MessagePage) {
  return page.messages.map(transformMessage)
}
```
