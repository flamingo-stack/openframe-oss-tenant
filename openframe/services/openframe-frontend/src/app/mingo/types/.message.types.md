Defines TypeScript interfaces and type guards for message objects used in the Flamingo/OpenFrame chat system, supporting both GraphQL API responses and core frontend message structures.

## Key Components

- **GraphQLMessage** - Interface for messages received from GraphQL API with dialog metadata
- **CoreMessage** - Interface for frontend message objects with role-based content
- **MessageConnection** - GraphQL connection pattern for paginated message results
- **MessagesResponse** - Complete GraphQL query response wrapper
- **MessagePage** - Processed page of messages with pagination info
- **Type Guards** - `isGraphQLMessage()` and `isCoreMessage()` for runtime type checking

## Usage Example

```typescript
import { GraphQLMessage, CoreMessage, isGraphQLMessage } from './message.types'

// Handle API response
const response: MessagesResponse = await fetchMessages()
const messages = response.data.messages.edges.map(edge => edge.node)

// Type-safe message processing
function processMessage(message: GraphQLMessage | CoreMessage) {
  if (isGraphQLMessage(message)) {
    // Handle GraphQL message structure
    return {
      id: message.id,
      chatType: message.chatType,
      data: message.messageData
    }
  } else {
    // Handle core message structure
    return {
      id: message.id,
      role: message.role,
      content: message.content
    }
  }
}

// Create core message
const userMessage: CoreMessage = {
  id: 'msg-1',
  role: 'user',
  content: { text: 'Hello Mingo!' },
  timestamp: new Date()
}
```
