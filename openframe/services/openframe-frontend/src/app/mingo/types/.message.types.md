Defines TypeScript type definitions for message-related data structures used across the Flamingo chat and ticketing system. Provides type safety for GraphQL message responses, core message formatting, and pagination handling.

## Key Components

- **GraphQLMessage**: Raw message data from GraphQL API with metadata like dialog ID, chat type, and owner information
- **CoreMessage**: Frontend-friendly message format with role, content, and display properties
- **MessageConnection**: GraphQL connection pattern for paginated message results
- **MessagesResponse**: Complete API response wrapper for message queries
- **MessagePage**: Simplified pagination structure for frontend consumption
- **Type Guards**: `isGraphQLMessage()` and `isCoreMessage()` functions for runtime type checking

## Usage Example

```typescript
// Type-safe message handling
function processMessage(message: GraphQLMessage | CoreMessage) {
  if (isGraphQLMessage(message)) {
    // Handle GraphQL message from API
    console.log(`Dialog: ${message.dialogId}, Type: ${message.chatType}`)
    return transformToCore(message)
  }
  
  if (isCoreMessage(message)) {
    // Handle frontend message display
    return renderMessage(message.content, message.role)
  }
}

// Pagination handling
function handleMessagePage(response: MessagesResponse) {
  const { messages, pageInfo } = response.data.messages
  
  return {
    items: messages.edges.map(edge => edge.node),
    hasMore: pageInfo.hasNextPage,
    cursor: pageInfo.endCursor
  }
}
```
