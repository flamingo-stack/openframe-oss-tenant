Type definitions for message data structures across different contexts in the OpenFrame chat system, providing interfaces for GraphQL responses, core message formats, and pagination data.

## Key Components

- **`GraphQLMessage`** - Server-side message structure with dialog metadata and owner information
- **`CoreMessage`** - Frontend message format with role-based content and assistant metadata  
- **`Message`** - Type alias for CoreMessage used throughout the application
- **`MessageConnection`** - GraphQL connection pattern for paginated message queries
- **`MessagesResponse`** - Complete GraphQL response wrapper for message data
- **`MessagePage`** - Simplified pagination structure for message lists
- **Type Guards** - `isGraphQLMessage()` and `isCoreMessage()` for runtime type checking

## Usage Example

```typescript
import { Message, GraphQLMessage, isGraphQLMessage } from './message.types'
import type { MessageContent } from '@flamingo-stack/openframe-frontend-core'

// Creating a core message for the frontend
const userMessage: Message = {
  id: 'msg-123',
  role: 'user',
  content: { text: 'Hello, how can I help?' } as MessageContent,
  timestamp: new Date(),
  avatar: null
}

// Type guard usage for message processing
function processMessage(message: GraphQLMessage | Message) {
  if (isGraphQLMessage(message)) {
    // Handle server message with dialog context
    console.log(`Dialog: ${message.dialogId}, Type: ${message.chatType}`)
  } else {
    // Handle frontend message
    console.log(`Role: ${message.role}, Content: ${message.content}`)
  }
}

// Working with paginated message responses
function handleMessagesResponse(response: MessagesResponse) {
  const { messages, pageInfo } = response.data.messages
  
  messages.edges.forEach(({ node }) => {
    console.log(`Message ${node.id} from ${node.owner.type}`)
  })
  
  if (pageInfo.hasNextPage) {
    // Load more messages using endCursor
    console.log(`Next page cursor: ${pageInfo.endCursor}`)
  }
}
```
