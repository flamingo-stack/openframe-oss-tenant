Zustand store for managing the detailed state of individual Mingo AI dialog sessions, including messages, typing indicators, and real-time updates.

## Key Components

- **useMingoDialogDetailsStore**: Main Zustand store hook for dialog details state management
- **Dialog State**: Current dialog ID, dialog node, and admin messages
- **Loading States**: Loading indicators for dialog and messages operations
- **Message Management**: Add, update, and filter messages with duplicate prevention
- **Typing Indicators**: Per-dialog typing states and typing message handling
- **Real-time Updates**: Add and update messages from WebSocket events
- **Pagination**: Cursor-based pagination for message history

## Usage Example

```typescript
import { useMingoDialogDetailsStore } from './stores/mingo-dialog-details-store'

function DialogDetailView() {
  const {
    currentDialog,
    adminMessages,
    isLoadingMessages,
    setCurrentDialogId,
    addRealtimeMessage,
    ensureTypingMessage,
    updateLastAssistantMessage
  } = useMingoDialogDetailsStore()

  // Select a dialog
  const handleSelectDialog = (dialogId: string) => {
    setCurrentDialogId(dialogId)
  }

  // Handle incoming WebSocket message
  const handleNewMessage = (message: Message) => {
    addRealtimeMessage(message)
  }

  // Handle AI response streaming
  const handleAIResponse = (dialogId: string, chunk: string) => {
    if (!useMingoDialogDetailsStore.getState().hasTypingMessage(dialogId)) {
      ensureTypingMessage(dialogId)
    }
    updateLastAssistantMessage(dialogId, chunk)
  }

  return (
    <div>
      {currentDialog && (
        <div>
          <h2>{currentDialog.title}</h2>
          {adminMessages.map(message => (
            <div key={message.id}>{message.messageData?.text}</div>
          ))}
          {isLoadingMessages && <div>Loading messages...</div>}
        </div>
      )}
    </div>
  )
}
```
