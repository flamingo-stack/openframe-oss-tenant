Zustand store managing the detailed view state and message handling for individual Mingo AI chat dialogs, including real-time message updates, typing indicators, and pagination.

## Key Components

- **State Management**: Current dialog selection, admin messages, loading states, and error handling
- **Message Operations**: Adding, updating, and filtering messages with duplicate prevention
- **Typing Indicators**: Managing typing states and temporary typing messages for real-time chat
- **Pagination**: Cursor-based message loading with hasMore flag
- **Real-time Updates**: Adding and updating messages from WebSocket or server-sent events

## Usage Example

```typescript
import { useMingoDialogDetailsStore } from './mingo-dialog-details-store'

function ChatDialog() {
  const {
    currentDialog,
    adminMessages,
    isLoadingMessages,
    setCurrentDialogId,
    addRealtimeMessage,
    ensureTypingMessage,
    updateLastAssistantMessage
  } = useMingoDialogDetailsStore()

  // Select a dialog to view
  const handleSelectDialog = (dialogId: string) => {
    setCurrentDialogId(dialogId)
  }

  // Handle incoming real-time message
  const handleNewMessage = (message: Message) => {
    addRealtimeMessage(message)
  }

  // Start typing indicator for AI response
  const handleAITyping = (dialogId: string) => {
    ensureTypingMessage(dialogId)
  }

  // Update streaming AI response
  const handleAIStream = (dialogId: string, chunk: string) => {
    updateLastAssistantMessage(dialogId, chunk)
  }

  return (
    <div>
      {isLoadingMessages && <Spinner />}
      {adminMessages.map(message => (
        <MessageComponent key={message.id} message={message} />
      ))}
    </div>
  )
}
```
