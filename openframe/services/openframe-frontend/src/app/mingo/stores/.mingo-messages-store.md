A Zustand store for managing Mingo chat dialog messages with real-time features, streaming support, and approval workflows. This store handles multiple dialogs, message history, typing indicators, unread counts, and message segment processing.

## Key Components

**State Management:**
- `messagesByDialog` - Map storing messages per dialog ID
- `activeDialogId` - Currently selected dialog
- `dialogs` - Array of available dialog nodes
- `typingStates` - Typing indicators per dialog
- `unreadCounts` - Unread message counts
- `streamingMessages` - Live streaming message state
- `segmentAccumulators` - Message segment processors for real-time updates

**Core Actions:**
- `setMessages/addMessage/updateMessage` - Message CRUD operations
- `setTyping/incrementUnread` - Real-time state updates
- `setStreamingMessage/updateStreamingMessageSegments` - Streaming support
- `getOrCreateAccumulator/updateAccumulatorApprovalStatus` - Approval workflow management

## Usage Example

```typescript
import { useMingoMessagesStore } from './mingo-messages-store'

function ChatComponent() {
  const {
    activeDialogId,
    getMessages,
    addMessage,
    setTyping,
    getStreamingMessage,
    updateStreamingMessageSegments
  } = useMingoMessagesStore()

  // Get messages for active dialog
  const messages = activeDialogId ? getMessages(activeDialogId) : []

  // Add a new message
  const handleSendMessage = (text: string) => {
    if (!activeDialogId) return
    
    const message: Message = {
      id: crypto.randomUUID(),
      role: 'user',
      content: text,
      timestamp: new Date().toISOString()
    }
    
    addMessage(activeDialogId, message)
  }

  // Handle streaming updates
  const handleStreamingSegments = (segments: MessageSegment[]) => {
    if (activeDialogId) {
      updateStreamingMessageSegments(activeDialogId, segments)
    }
  }

  return (
    <div>
      {messages.map(message => (
        <MessageComponent key={message.id} message={message} />
      ))}
    </div>
  )
}
```
