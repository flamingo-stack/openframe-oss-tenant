A Zustand store for managing chat messages, dialogs, and real-time communication state in the Mingo AI assistant interface. This store provides unified message management across multiple dialogs with support for streaming messages, approval workflows, and real-time indicators.

## Key Components

**State Management:**
- `messagesByDialog` - Map storing messages grouped by dialog ID
- `activeDialogId` - Currently selected dialog
- `dialogs` - Array of available dialog nodes
- `typingStates` - Real-time typing indicators per dialog
- `unreadCounts` - Unread message counts per dialog
- `streamingMessages` - Currently streaming messages
- `segmentAccumulators` - Message segment processors for streaming content

**Core Actions:**
- `setMessages/addMessage/updateMessage` - Message CRUD operations
- `setActiveDialogId` - Switch between dialogs
- `setTyping/incrementUnread` - Real-time state updates
- `updateStreamingMessageSegments` - Handle streaming message updates
- `getOrCreateAccumulator` - Manage message segment accumulators

## Usage Example

```typescript
import { useMingoMessagesStore } from './mingo-messages-store'

function ChatInterface() {
  const {
    activeDialogId,
    getMessages,
    addMessage,
    setTyping,
    getStreamingMessage,
    updateStreamingMessageSegments
  } = useMingoMessagesStore()

  // Get messages for active dialog
  const messages = activeDialogId ? getMessages(activeDialogId) : []

  // Add a new message
  const sendMessage = (content: string) => {
    if (!activeDialogId) return
    
    const message = {
      id: crypto.randomUUID(),
      role: 'user' as const,
      content,
      timestamp: new Date().toISOString()
    }
    
    addMessage(activeDialogId, message)
  }

  // Handle streaming updates
  const handleStreamingSegments = (segments: MessageSegment[]) => {
    if (activeDialogId) {
      updateStreamingMessageSegments(activeDialogId, segments)
    }
  }

  return (
    <div>
      {messages.map(message => (
        <MessageComponent key={message.id} message={message} />
      ))}
    </div>
  )
}
```
