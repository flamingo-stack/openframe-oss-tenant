A Zustand store for managing Mingo AI chat messages, dialog state, and real-time communication features. This store handles message persistence, streaming updates, approval workflows, and UI state management across multiple chat dialogs.

## Key Components

- **MingoMessagesStore Interface**: Comprehensive store interface defining message management, dialog state, real-time features, and UI controls
- **Message Management**: CRUD operations for messages across multiple dialogs with Map-based storage
- **Real-time State**: Typing indicators, unread counts, and streaming message handling
- **Segment Accumulators**: Message segment processing for tool executions and approval requests
- **Loading & Error States**: UI state management for different async operations
- **Pagination Support**: Cursor-based message pagination with newest message tracking

## Usage Example

```typescript
import { useMingoMessagesStore } from './mingo-messages-store'

// In a React component
function ChatInterface() {
  const {
    activeDialogId,
    getMessages,
    addMessage,
    setTyping,
    getUnread,
    setStreamingMessage,
    getOrCreateAccumulator
  } = useMingoMessagesStore()

  // Get messages for current dialog
  const messages = activeDialogId ? getMessages(activeDialogId) : []
  
  // Add a new message
  const sendMessage = (dialogId: string, content: string) => {
    addMessage(dialogId, {
      id: crypto.randomUUID(),
      role: 'user',
      content: content,
      timestamp: new Date().toISOString()
    })
  }

  // Handle streaming updates
  const handleStreamingUpdate = (dialogId: string, segments: MessageSegment[]) => {
    const accumulator = getOrCreateAccumulator(dialogId, {
      onApprove: (requestId) => handleApproval(requestId, 'approved'),
      onReject: (requestId) => handleApproval(requestId, 'rejected')
    })
    
    // Process streaming segments through accumulator
    useMingoMessagesStore.getState().updateStreamingMessageSegments(dialogId, segments)
  }

  return (
    <div>
      {messages.map(message => (
        <MessageComponent key={message.id} message={message} />
      ))}
    </div>
  )
}
```
