A Zustand store managing message state and real-time interactions for the Mingo AI assistant, handling dialogs, streaming messages, and approval workflows with comprehensive message segment accumulation.

## Key Components

- **Message Storage**: Map-based storage for messages organized by dialog ID
- **Dialog Management**: Active dialog tracking and dialog node management
- **Real-time State**: Typing indicators, unread counts, and streaming message handling
- **Message Segments**: Integration with `MessageSegmentAccumulator` for processing streaming content
- **Approval System**: Approval request tracking and status updates within message content
- **Loading & Error States**: Comprehensive state management for async operations
- **Pagination**: Cursor-based pagination support for message history

## Usage Example

```typescript
import { useMingoMessagesStore } from './mingo-messages-store'

// Component using the store
function ChatInterface() {
  const {
    messagesByDialog,
    activeDialogId,
    setActiveDialogId,
    addMessage,
    getTyping,
    setTyping,
    getOrCreateAccumulator
  } = useMingoMessagesStore()

  // Switch to a dialog
  const selectDialog = (dialogId: string) => {
    setActiveDialogId(dialogId)
    // Reset unread count when dialog becomes active
    useMingoMessagesStore.getState().resetUnread(dialogId)
  }

  // Add new message
  const sendMessage = (content: string) => {
    if (activeDialogId) {
      addMessage(activeDialogId, {
        id: crypto.randomUUID(),
        role: 'user',
        content,
        timestamp: new Date().toISOString()
      })
    }
  }

  // Handle streaming message segments
  const handleStreamingSegments = (segments: MessageSegment[]) => {
    if (activeDialogId) {
      const accumulator = getOrCreateAccumulator(activeDialogId, {
        onApprove: (requestId) => handleApproval(requestId, 'approved'),
        onReject: (requestId) => handleApproval(requestId, 'rejected')
      })
      
      useMingoMessagesStore.getState()
        .updateStreamingMessageSegments(activeDialogId, segments)
    }
  }

  return (
    <div>
      {/* Chat UI implementation */}
    </div>
  )
}
```
