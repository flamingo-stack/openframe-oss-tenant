A Zustand store that manages Mingo AI chat dialog state, including messages, typing indicators, streaming responses, and approval workflows across multiple conversations.

## Key Components

**State Management:**
- `messagesByDialog` - Map storing messages per dialog ID
- `activeDialogId` - Currently selected dialog
- `dialogs` - Array of dialog nodes
- `typingStates` - Real-time typing indicators per dialog
- `unreadCounts` - Unread message counts per dialog
- `streamingMessages` - Active streaming messages per dialog
- `segmentAccumulators` - Message segment processors per dialog

**Core Actions:**
- `setMessages/addMessage/updateMessage` - Message CRUD operations
- `setTyping/getTyping` - Typing indicator management
- `setStreamingMessage/updateStreamingMessageSegments` - Real-time streaming
- `getOrCreateAccumulator` - Segment accumulator management
- `updateApprovalStatusInMessages` - Approval workflow handling

## Usage Example

// In a chat component

  const {
    activeDialogId,
    setActiveDialogId,
    getMessages,
    addMessage,
    setTyping,
    getOrCreateAccumulator
  } = useMingoMessagesStore()
  
  // Switch to a dialog
  const selectDialog = (dialogId: string) => {
    setActiveDialogId(dialogId)
  }
  
  // Add a user message
  const sendMessage = (content: string) => {
    if (!activeDialogId) return
    
    const message = {
      id: generateId(),
      role: 'user' as const,
      content,
      timestamp: Date.now()
    }
    
    addMessage(activeDialogId, message)
  }
  
  // Handle streaming responses
  const handleStreamingSegments = (segments: MessageSegment[]) => {
    if (!activeDialogId) return
    
    const accumulator = getOrCreateAccumulator(activeDialogId, {
      onApprove: (requestId) => handleApproval(requestId, 'approved'),
      onReject: (requestId) => handleApproval(requestId, 'rejected')
    })
    
    updateStreamingMessageSegments(activeDialogId, segments)
  }
  
  return <ChatUI messages={getMessages(activeDialogId || '')} />
}
