A Zustand store for managing Mingo chat conversations, real-time messaging state, and streaming message processing with approval workflows.

## Key Components

- **MingoMessagesStore**: Main store interface with complete message and dialog management
- **useMingoMessagesStore**: Zustand hook for accessing the store
- **Message Management**: CRUD operations for messages across multiple dialogs
- **Real-time State**: Typing indicators, unread counts, and streaming message handling
- **Segment Accumulators**: Processing streaming message segments with approval requests
- **Loading & Error States**: UI state management for async operations

## Usage Example

```typescript
import { useMingoMessagesStore } from './mingo-messages-store'

function ChatComponent() {
  const {
    messagesByDialog,
    activeDialogId,
    addMessage,
    setTyping,
    getOrCreateAccumulator,
    updateStreamingMessageSegments
  } = useMingoMessagesStore()

  // Get messages for active dialog
  const currentMessages = activeDialogId 
    ? messagesByDialog.get(activeDialogId) || []
    : []

  // Handle new message
  const sendMessage = (content: string) => {
    if (!activeDialogId) return
    
    const message = {
      id: crypto.randomUUID(),
      role: 'user' as const,
      content,
      timestamp: Date.now()
    }
    
    addMessage(activeDialogId, message)
  }

  // Handle streaming response with approval requests
  const handleStreamingSegments = (segments: MessageSegment[]) => {
    if (!activeDialogId) return
    
    const accumulator = getOrCreateAccumulator(activeDialogId, {
      onApprove: (requestId) => console.log('Approved:', requestId),
      onReject: (requestId) => console.log('Rejected:', requestId)
    })
    
    updateStreamingMessageSegments(activeDialogId, segments)
  }

  return (
    <div>
      {currentMessages.map(message => (
        <div key={message.id}>{message.content}</div>
      ))}
    </div>
  )
}
```
