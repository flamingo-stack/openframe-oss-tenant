Centralized Zustand store for managing Mingo AI chat dialogs, messages, and real-time communication state. Handles message streaming, approval requests, typing indicators, and unread counts across multiple dialog sessions.

## Key Components

- **MingoMessagesStore**: Main store interface with dialog and message management
- **useMingoMessagesStore**: Zustand store hook with devtools integration
- **Message Management**: CRUD operations for messages across dialogs
- **Real-time State**: Typing indicators, unread counts, and streaming message handling
- **Segment Accumulators**: Processing and accumulating message segments during streaming
- **Pagination & Loading States**: Dialog loading, message pagination, and error handling

## Usage Example

```typescript
import { useMingoMessagesStore } from './mingo-messages-store'

// Component using the store
function ChatInterface() {
  const {
    activeDialogId,
    setActiveDialogId,
    getMessages,
    addMessage,
    setTyping,
    getUnread,
    resetUnread
  } = useMingoMessagesStore()

  // Switch to a dialog
  const selectDialog = (dialogId: string) => {
    setActiveDialogId(dialogId)
    resetUnread(dialogId) // Mark as read
  }

  // Add new message
  const sendMessage = (content: string) => {
    if (!activeDialogId) return
    
    const message = {
      id: crypto.randomUUID(),
      role: 'user' as const,
      content,
      timestamp: new Date().toISOString()
    }
    
    addMessage(activeDialogId, message)
  }

  // Get messages for current dialog
  const messages = activeDialogId ? getMessages(activeDialogId) : []
  const unreadCount = activeDialogId ? getUnread(activeDialogId) : 0

  return (
    <div>
      {messages.map(msg => (
        <div key={msg.id}>{msg.content}</div>
      ))}
      {unreadCount > 0 && <span>({unreadCount} unread)</span>}
    </div>
  )
}

// Streaming message handling
const { 
  setStreamingMessage, 
  updateStreamingMessageSegments,

} = useMingoMessagesStore()

// Handle streaming segments
const handleStreamSegments = (dialogId: string, segments: MessageSegment[]) => {
  const accumulator = getOrCreateAccumulator(dialogId, {

  })
  
  updateStreamingMessageSegments(dialogId, segments)
}
```
