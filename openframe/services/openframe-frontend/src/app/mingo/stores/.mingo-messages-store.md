A Zustand store for managing Mingo AI chat messages, dialogs, and real-time communication state across the application. This store handles message persistence, streaming updates, approval workflows, and dialog management with comprehensive state tracking.

## Key Components

### State Management
- **messagesByDialog**: Map storing messages organized by dialog ID
- **activeDialogId**: Currently selected dialog
- **dialogs**: Array of available dialog nodes
- **typingStates**: Real-time typing indicators per dialog
- **unreadCounts**: Unread message counts per dialog
- **streamingMessages**: Active streaming messages being received
- **segmentAccumulators**: Message segment processors for real-time updates

### Core Actions
- **Message CRUD**: `addMessage`, `updateMessage`, `removeMessage`, `setMessages`
- **Dialog Management**: `setActiveDialogId`, `setDialogs`, `clearDialog`
- **Real-time Updates**: `setTyping`, `incrementUnread`, `resetUnread`
- **Streaming Support**: `setStreamingMessage`, `updateStreamingMessageSegments`
- **Approval Workflow**: `updateApprovalStatusInMessages`, `updateAccumulatorApprovalStatus`

### Loading & Error States
- **Loading indicators**: `isLoadingDialog`, `isLoadingMessages`, `isCreatingDialog`
- **Error handling**: `dialogError`, `messagesError`
- **Pagination**: `hasMoreMessages`, `messagesCursor`, `newestMessageCursor`

// Component using the store
function ChatInterface() {
  const {
    activeDialogId,
    getMessages,
    addMessage,
    setTyping,
    getStreamingMessage,
    getOrCreateAccumulator
  } = useMingoMessagesStore()

  // Get messages for active dialog
  const messages = activeDialogId ? getMessages(activeDialogId) : []
  
  // Add a new message
  const sendMessage = (content: string) => {
    if (!activeDialogId) return
    
    const newMessage = {
      id: generateId(),
      role: 'user' as const,
      content,
      timestamp: Date.now()
    }
    
    addMessage(activeDialogId, newMessage)
  }

  // Handle streaming updates
  const handleStreamingUpdate = (segments: MessageSegment[]) => {
    if (!activeDialogId) return
    
    const accumulator = getOrCreateAccumulator(activeDialogId, {
      onApprove: (requestId) => console.log('Approved:', requestId),
      onReject: (requestId) => console.log('Rejected:', requestId)
    })
    
    // Process streaming segments
    updateStreamingMessageSegments(activeDialogId, segments)
  }

  return (
    <div>
      {messages.map(message => (
        <MessageComponent key={message.id} message={message} />
      ))}
    </div>
  )
}
