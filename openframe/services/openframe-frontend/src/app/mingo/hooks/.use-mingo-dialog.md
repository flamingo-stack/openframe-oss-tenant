A custom React hook that manages Mingo AI chat dialogs, providing functionality to create new conversations and send messages with automatic error handling and toast notifications.

## Key Components

- **useMingoDialog**: Main hook that orchestrates dialog creation and message sending
- **createDialogMutation**: TanStack Query mutation for creating new admin-type dialogs
- **sendMessageMutation**: TanStack Query mutation for sending messages to existing dialogs
- **Dialog State Management**: Tracks current dialog ID using both state and ref for persistence
- **Error Handling**: Comprehensive error handling with user-friendly toast notifications
- **Auto-Creation Logic**: Automatically creates dialogs when sending messages without an active dialog

## Usage Example

```typescript
import { useMingoDialog } from './hooks/use-mingo-dialog'

function ChatInterface() {
  const {
    currentDialogId,
    isCreatingDialog,
    isSendingMessage,
    sendMessage,
    createDialog,
    resetDialog,
    hasDialog
  } = useMingoDialog()

  const handleSendMessage = async (message: string) => {
    const success = await sendMessage(message)
    if (success) {
      console.log('Message sent successfully')
    }
  }

  const handleNewChat = async () => {
    resetDialog() // Clear current dialog
    const newDialogId = await createDialog()
    if (newDialogId) {
      console.log(`New dialog created: ${newDialogId}`)
    }
  }

  return (
    <div>
      <button onClick={handleNewChat} disabled={isCreatingDialog}>
        {isCreatingDialog ? 'Creating...' : 'New Chat'}
      </button>
      
      <input
        onKeyDown={(e) => {
          if (e.key === 'Enter') {
            handleSendMessage(e.currentTarget.value)
            e.currentTarget.value = ''
          }
        }}
        disabled={isSendingMessage}
        placeholder="Type your message..."
      />
      
      {hasDialog && <p>Active dialog: {currentDialogId}</p>}
    </div>
  )
}
```
