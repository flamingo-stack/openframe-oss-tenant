A React hook that provides a complete chat interface for interacting with Mingo AI, handling real-time message streaming, approval workflows, and dialog management with optimistic updates and error handling.

## Key Components

- **useMingoChat**: Main hook providing chat functionality with message history, real-time processing, and dialog management
- **ProcessedMessage**: Interface for formatted chat messages with role, content, and metadata
- **ApprovalHandlers**: Interface for handling approval/rejection actions in chat workflows
- **Real-time Processing**: Integration with NATS messaging for live chat updates and streaming responses
- **Message Store Integration**: Uses Zustand store for persistent message state management

## Usage Example

```typescript
import { useMingoChat } from './hooks/use-mingo-chat'

function ChatInterface({ dialogId }: { dialogId: string | null }) {
  const approvalHandlers = {
    handleApprove: (requestId?: string) => {
      console.log('Approved request:', requestId)
    },
    handleReject: (requestId?: string) => {
      console.log('Rejected request:', requestId)
    },
    approvalStatuses: {}
  }

  const {
    messages,
    isLoading,
    isTyping,
    approvals,
    createDialog,
    sendMessage,
    processChunk
  } = useMingoChat(dialogId, approvalHandlers)

  const handleSend = async (message: string) => {
    if (!dialogId) {
      const newDialogId = await createDialog()
      if (newDialogId) {
        await sendMessage(message)
      }
    } else {
      await sendMessage(message)
    }
  }

  return (
    <div className="chat-interface">
      {messages.map(msg => (
        <div key={msg.id} className={`message ${msg.role}`}>
          <strong>{msg.name}:</strong> {msg.content}
        </div>
      ))}
      {isTyping && <div>Mingo is typing...</div>}
      {approvals.length > 0 && (
        <div className="approvals">
          Pending approvals: {approvals.length}
        </div>
      )}
    </div>
  )
}
```
