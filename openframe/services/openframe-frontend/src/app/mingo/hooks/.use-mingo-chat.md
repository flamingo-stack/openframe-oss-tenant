A React hook that manages Mingo AI chat functionality with real-time message processing, approval handling, and dialog management. It integrates with Zustand store for state management and provides WebSocket-based streaming capabilities.

## Key Components

- **useMingoChat** - Main hook that returns chat state and actions
- **ProcessedMessage** - Interface for formatted chat messages
- **UseMingoChat** - Return type interface with messages, actions, and state
- **ApprovalHandlers** - Interface for handling approval requests
- **Real-time processors** - Callbacks for streaming message updates
- **Dialog management** - Functions for creating and managing chat sessions

## Usage Example

```typescript
import { useMingoChat } from './use-mingo-chat'

function ChatComponent() {
  const [dialogId, setDialogId] = useState<string | null>(null)
  
  const approvalHandlers = {
    handleApprove: (requestId?: string) => {
      console.log('Approved:', requestId)
    },
    handleReject: (requestId?: string) => {
      console.log('Rejected:', requestId)
    },
    approvalStatuses: {}
  }
  
  const {
    messages,
    isLoading,
    isTyping,
    approvals,
    createDialog,
    sendMessage,
    processChunk
  } = useMingoChat(dialogId, approvalHandlers)
  
  const handleCreateChat = async () => {
    const newDialogId = await createDialog()
    if (newDialogId) {
      setDialogId(newDialogId)
    }
  }
  
  const handleSendMessage = async (content: string) => {
    const success = await sendMessage(content)
    if (!success) {
      console.error('Failed to send message')
    }
  }
  
  return (
    <div>
      {!dialogId ? (
        <button onClick={handleCreateChat}>Start Chat</button>
      ) : (
        <>
          <div>
            {messages.map(msg => (
              <div key={msg.id}>
                <strong>{msg.name}:</strong> {msg.content}
              </div>
            ))}
            {isTyping && <div>Mingo is typing...</div>}
          </div>
          {approvals.length > 0 && (
            <div>Pending approvals: {approvals.length}</div>
          )}
        </>
      )}
    </div>
  )
}
```
