A React hook for managing Mingo AI chat functionality, providing message handling, dialog creation, and approval system integration with optimistic updates and real-time typing states.

## Key Components

- **useMingoChat** - Main hook function that manages chat state and operations
- **ProcessedMessage** - Interface for formatted chat messages with role, content, and metadata
- **UseMingoChat** - Return type interface defining available methods and state
- **createDialog** - Async function to create new chat dialogs
- **sendMessage** - Async function to send messages with optimistic updates
- **approvals** - Computed array of pending approval requests from message segments

## Usage Example

```typescript
import { useMingoChat } from './hooks/use-mingo-chat'

function ChatComponent() {
  const dialogId = "dialog-123"
  const {
    messages,
    isLoading,
    isTyping,
    approvals,
    createDialog,
    sendMessage
  } = useMingoChat(dialogId)

  const handleSend = async (content: string) => {
    const success = await sendMessage(content)
    if (success) {
      console.log('Message sent successfully')
    }
  }

  const handleCreateDialog = async () => {
    const newDialogId = await createDialog()
    if (newDialogId) {
      // Navigate to new dialog
    }
  }

  return (
    <div>
      {messages.map(msg => (
        <div key={msg.id} className={`message-${msg.role}`}>
          <strong>{msg.name}:</strong> {msg.content}
        </div>
      ))}
      {isTyping && <div>Mingo is typing...</div>}
      {approvals.length > 0 && (
        <div>Pending approvals: {approvals.length}</div>
      )}
    </div>
  )
}
```
