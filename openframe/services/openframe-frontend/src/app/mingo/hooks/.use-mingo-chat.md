This hook provides a comprehensive interface for interacting with the Mingo AI assistant, managing real-time chat functionality including message processing, streaming responses, and approval workflows.

## Key Components

- **Message Management**: Processes and filters chat messages, handling different roles (user/assistant/error)
- **Real-time Streaming**: Integrates with `useRealtimeChunkProcessor` for live response streaming
- **Approval System**: Manages pending approvals and provides handlers for approve/reject actions
- **Dialog Operations**: Creates new chat sessions and sends messages with optimistic updates
- **State Management**: Tracks typing indicators, loading states, and message accumulation
- **Error Handling**: Gracefully handles streaming errors and API failures

## Usage Example

```typescript
import { useMingoChat } from './hooks/use-mingo-chat'

function ChatComponent() {
  const [dialogId, setDialogId] = useState<string | null>(null)
  
  const approvalHandlers = {
    handleApprove: (requestId?: string) => {
      // Handle approval logic
    },
    handleReject: (requestId?: string) => {
      // Handle rejection logic
    },
    approvalStatuses: {}
  }
  
  const {
    messages,
    isLoading,
    createDialog,
    sendMessage,
    approvals,
    processChunk,
    isTyping
  } = useMingoChat(dialogId, approvalHandlers)
  
  const handleCreateChat = async () => {
    const newDialogId = await createDialog()
    if (newDialogId) setDialogId(newDialogId)
  }
  
  const handleSendMessage = async (content: string) => {
    await sendMessage(content)
  }
  
  return (
    <div>
      {messages.map(msg => (
        <div key={msg.id}>{msg.content}</div>
      ))}
      {isTyping && <div>Mingo is typing...</div>}
    </div>
  )
}
```
