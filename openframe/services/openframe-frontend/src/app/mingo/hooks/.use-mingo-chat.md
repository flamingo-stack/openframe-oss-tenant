A React hook for managing Mingo AI chat conversations with message handling, dialog creation, and approval workflows. This hook integrates with the Mingo Messages Store and provides a complete chat interface for the Mingo assistant.

## Key Components

- **Messages Management**: Processes and filters chat messages, excluding pending approvals from display
- **Dialog Operations**: Creates new chat dialogs and manages conversation state
- **Message Sending**: Handles optimistic updates and error recovery for message transmission
- **Approval System**: Extracts and manages pending approval requests from message segments
- **Typing Indicators**: Tracks and displays assistant typing states
- **State Management**: Integrates with Zustand store and React Query for data persistence

## Usage Example

```typescript
import { useMingoChat } from './hooks/use-mingo-chat'

function MingoChatComponent() {
  const { 
    messages, 
    sendMessage, 
    createDialog, 
    approvals, 
    isTyping 
  } = useMingoChat('dialog-123')

  const handleSendMessage = async (content: string) => {
    const success = await sendMessage(content)
    if (!success) {
      console.error('Failed to send message')
    }
  }

  const handleCreateDialog = async () => {
    const newDialogId = await createDialog()
    if (newDialogId) {
      // Navigate to new dialog
    }
  }

  return (
    <div>
      {messages.map(msg => (
        <div key={msg.id}>{msg.content}</div>
      ))}
      {approvals.length > 0 && (
        <div>Pending approvals: {approvals.length}</div>
      )}
      {isTyping && <div>Mingo is typing...</div>}
    </div>
  )
}
```
