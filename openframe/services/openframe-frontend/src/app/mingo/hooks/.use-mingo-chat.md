A React hook for managing Mingo AI chat functionality, providing real-time messaging, approval workflows, and dialog management with optimistic updates and error handling.

## Key Components

- **useMingoChat**: Main hook that manages chat state, messaging, and real-time processing
- **ProcessedMessage**: Interface for formatted chat messages with role, content, and metadata
- **ApprovalHandlers**: Interface for handling approval requests from the AI assistant
- **Real-time processing**: Integration with NATS for streaming message chunks
- **Dialog management**: Create new chat sessions and manage message history
- **Optimistic updates**: Immediate UI feedback while API calls are in progress

## Usage Example

```typescript
import { useMingoChat } from './hooks/use-mingo-chat'

function ChatComponent({ dialogId }: { dialogId: string | null }) {
  const approvalHandlers = {
    handleApprove: (requestId?: string) => {
      console.log('Approved:', requestId)
    },
    handleReject: (requestId?: string) => {
      console.log('Rejected:', requestId)
    },
    approvalStatuses: {}
  }

  const {
    messages,
    isLoading,
    isTyping,
    approvals,
    createDialog,
    sendMessage,
    processChunk
  } = useMingoChat(dialogId, approvalHandlers)

  const handleSend = async (content: string) => {
    const success = await sendMessage(content)
    if (!success) {
      console.error('Failed to send message')
    }
  }

  const handleNewChat = async () => {
    const newDialogId = await createDialog()
    if (newDialogId) {
      // Navigate to new dialog
    }
  }

  return (
    <div>
      {messages.map(msg => (
        <div key={msg.id}>
          <strong>{msg.name}:</strong> {msg.content}
        </div>
      ))}
      {isTyping && <div>Mingo is typing...</div>}
      {approvals.length > 0 && <div>Pending approvals: {approvals.length}</div>}
    </div>
  )
}
```
