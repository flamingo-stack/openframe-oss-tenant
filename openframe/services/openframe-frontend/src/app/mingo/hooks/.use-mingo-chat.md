This hook provides a complete real-time chat interface for Mingo AI conversations, handling message streaming, approval workflows, and dialog management with optimistic updates and error handling.

## Key Components

- **Message Management**: Stores and processes chat messages with real-time updates
- **Real-time Streaming**: Handles chunked message processing via WebSocket connections
- **Approval System**: Manages approval requests for sensitive operations
- **Dialog Operations**: Creates new chat sessions and sends messages
- **State Management**: Tracks typing indicators, loading states, and message history
- **Error Handling**: Gracefully handles errors with user feedback

## Usage Example

```typescript
import { useMingoChat } from './hooks/use-mingo-chat'

function ChatInterface() {
  const [currentDialogId, setCurrentDialogId] = useState<string | null>(null)
  
  const approvalHandlers = {
    handleApprove: (requestId?: string) => {
      // Handle approval logic
      console.log('Approved:', requestId)
    },
    handleReject: (requestId?: string) => {
      // Handle rejection logic
      console.log('Rejected:', requestId)
    },
    approvalStatuses: {}
  }
  
  const {
    messages,
    isLoading,
    createDialog,
    sendMessage,
    approvals,
    isTyping,
    processChunk
  } = useMingoChat(currentDialogId, approvalHandlers)
  
  const handleNewChat = async () => {
    const newDialogId = await createDialog()
    if (newDialogId) {
      setCurrentDialogId(newDialogId)
    }
  }
  
  const handleSendMessage = async (content: string) => {
    const success = await sendMessage(content)
    if (!success) {
      console.error('Failed to send message')
    }
  }
  
  return (
    <div className="chat-container">
      <button onClick={handleNewChat}>New Chat</button>
      
      {messages.map(message => (
        <div key={message.id} className={`message ${message.role}`}>
          <strong>{message.name}:</strong>
          <span>{typeof message.content === 'string' ? message.content : '[Rich Content]'}</span>
        </div>
      ))}
      
      {isTyping && <div>Mingo is typing...</div>}
      
      {approvals.length > 0 && (
        <div className="approvals">
          <h3>Pending Approvals: {approvals.length}</h3>
        </div>
      )}
      
      <input 
        onKeyPress={(e) => {
          if (e.key === 'Enter') {
            handleSendMessage(e.target.value)
            e.target.value = ''
          }
        }}
        disabled={isTyping}
      />
    </div>
  )
}
```
