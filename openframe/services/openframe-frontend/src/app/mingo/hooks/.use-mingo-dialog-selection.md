React hook for managing Mingo dialog selection, message loading, and approval workflows with infinite scrolling and real-time status updates.

## Key Components

- **Dialog Selection**: Mutation-based dialog switching with loading state management
- **Message Loading**: Infinite query for paginated message fetching with automatic pagination
- **Approval System**: Handlers for approving/rejecting requests with optimistic updates
- **State Management**: Integration with Zustand store for dialog and message persistence
- **Error Handling**: Toast notifications for approval failures and query errors

## Usage Example

```typescript
import { useMingoDialogSelection } from './hooks/use-mingo-dialog-selection'

function MingoDialogComponent() {
  const {
    selectDialog,
    isSelectingDialog,
    isLoadingMessages,
    handleApprove,
    handleReject,
    approvalStatuses,
    rawMessagesCount
  } = useMingoDialogSelection()

  const handleDialogClick = (dialogId: string) => {
    selectDialog(dialogId)
  }

  const handleApproveRequest = (requestId: string) => {
    handleApprove(requestId)
  }

  return (
    <div>
      <button 
        onClick={() => handleDialogClick('dialog-123')}
        disabled={isSelectingDialog}
      >
        Select Dialog
      </button>
      
      {isLoadingMessages ? (
        <div>Loading messages...</div>
      ) : (
        <div>
          {rawMessagesCount} messages loaded
          <button onClick={() => handleApproveRequest('req-456')}>
            Approve Request
          </button>
        </div>
      )}
    </div>
  )
}
```
