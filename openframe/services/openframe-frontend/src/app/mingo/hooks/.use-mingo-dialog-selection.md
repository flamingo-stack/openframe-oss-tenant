A React hook for managing Mingo dialog selection and message processing, including infinite scrolling, message approval/rejection workflows, and real-time state synchronization.

## Key Components

- **Dialog Management**: Handles active dialog selection and switching with optimistic state updates
- **Message Processing**: Converts GraphQL messages to standardized format with approval status tracking
- **Infinite Scrolling**: Implements paginated message loading with automatic next page fetching
- **Approval Workflow**: Provides approve/reject handlers for request processing with error handling
- **State Synchronization**: Manages approval statuses and message updates across the application

## Usage Example

```typescript
import { useMingoDialogSelection } from './use-mingo-dialog-selection'

function MingoDialogInterface() {
  const {
    selectDialog,
    isSelectingDialog,
    isLoadingMessages,
    handleApprove,
    handleReject,
    approvalStatuses,
    rawMessagesCount,
    dialogError,
    refetchMessages
  } = useMingoDialogSelection()

  const handleDialogSelect = (dialogId: string) => {
    selectDialog(dialogId)
  }

  const handleRequestApproval = (requestId: string) => {
    handleApprove(requestId)
  }

  return (
    <div>
      {isLoadingMessages && <div>Loading messages...</div>}
      {dialogError && <div>Error: {dialogError}</div>}
      <div>Messages loaded: {rawMessagesCount}</div>
      
      <button 
        onClick={() => handleRequestApproval('req-123')}
        disabled={approvalStatuses['req-123'] === 'approved'}
      >
        Approve Request
      </button>
    </div>
  )
}
```
