React hook for managing real-time NATS WebSocket subscriptions to Mingo AI chat dialogs with automatic message catchup and multi-dialog support.

## Key Components

- **`useMingoRealtimeSubscription`** - Main hook for managing multiple dialog subscriptions with connection state tracking
- **`DialogSubscription`** - Component that handles individual dialog NATS subscriptions and message processing
- **Connection Management** - Handles WebSocket URLs, authentication tokens, and connection states
- **Dialog State Tracking** - Maintains subscription status, connection state, and catchup status per dialog
- **Token Management** - Monitors localStorage for auth token changes in development mode

## Usage Example

```typescript
// Basic subscription management
function MingoChat() {
  const [activeDialogId, setActiveDialogId] = useState<string | null>(null)
  
  const {
    subscribeToDialog,
    unsubscribeFromDialog,
    getSubscriptionState,
    subscribedDialogs,
    connectionState
  } = useMingoRealtimeSubscription(activeDialogId, {
    onChunkReceived: (dialogId, chunk, messageType) => {
      console.log(`Received chunk for dialog ${dialogId}:`, chunk)
    }
  })

  // Subscribe to a new dialog
  const handleOpenDialog = (dialogId: string) => {
    setActiveDialogId(dialogId)
    subscribeToDialog(dialogId) // Auto-subscribes to active dialog
  }

  // Render dialog subscriptions
  return (
    <div>
      {Array.from(subscribedDialogs).map(dialogId => (
        <DialogSubscription
          key={dialogId}
          dialogId={dialogId}
          isActive={dialogId === activeDialogId}
          processChunk={(targetDialogId, chunk, messageType) => {
            // Handle incoming chunks
            updateMessages(targetDialogId, chunk)
          }}
        />
      ))}
      <div>Connection: {connectionState}</div>
    </div>
  )
}
```
