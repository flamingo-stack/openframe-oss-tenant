A React hook that manages real-time NATS WebSocket subscriptions for Mingo AI chat dialogs, providing unified subscription management across multiple conversations.

## Key Components

- **useMingoRealtimeSubscription**: Main hook for managing multiple dialog subscriptions
- **DialogSubscription**: Component for individual dialog subscription handling
- **getApiBaseUrl()**: Utility to resolve the API base URL from environment or window
- **getAccessToken()**: Utility to retrieve auth token from localStorage
- **DialogSubscriptionState**: Interface defining subscription connection states
- **UseMingoRealtimeSubscriptionOptions**: Configuration options with chunk callback

## Usage Example

```typescript
import { useMingoRealtimeSubscription, DialogSubscription } from './use-mingo-realtime-subscription'
import { ChunkData, NatsMessageType } from '@flamingo-stack/openframe-frontend-core'

function MingoChat() {
  const {
    subscribeToDialog,
    unsubscribeFromDialog,
    subscribedDialogs,
    connectionState
  } = useMingoRealtimeSubscription('dialog-123', {
    onChunkReceived: (dialogId, chunk, messageType) => {
      console.log(`Received chunk for ${dialogId}:`, chunk)
    }
  })

  const handleJoinDialog = (dialogId: string) => {
    subscribeToDialog(dialogId)
  }

  return (
    <div>
      <p>Connection: {connectionState}</p>
      {Array.from(subscribedDialogs).map(dialogId => (
        <DialogSubscription
          key={dialogId}
          dialogId={dialogId}
          isActive={dialogId === 'dialog-123'}
          processChunk={(targetDialogId, chunk, messageType) => {
            // Handle real-time chunks
            console.log('Processing chunk:', chunk)
          }}
        />
      ))}
    </div>
  )
}
```
