A React hook for managing real-time NATS subscriptions to Mingo chat dialogs, providing WebSocket-based message streaming with automatic catchup functionality.

## Key Components

- **useMingoRealtimeSubscription**: Main hook for managing multiple dialog subscriptions with connection state tracking
- **DialogSubscription**: Component that handles individual dialog NATS subscriptions and message processing
- **getApiBaseUrl/getAccessToken**: Utility functions for configuration and authentication
- **DialogSubscriptionState**: Interface tracking subscription, connection, and catchup status
- **MINGO_TOPICS**: Supported NATS message types (`admin-message`)

## Usage Example

```typescript
import { useMingoRealtimeSubscription, DialogSubscription } from './use-mingo-realtime-subscription'

function ChatComponent() {
  const {
    subscribeToDialog,
    unsubscribeFromDialog,
    getSubscriptionState,
    subscribedDialogs,
    connectionState
  } = useMingoRealtimeSubscription('dialog-123', {
    onChunkReceived: (dialogId, chunk, messageType) => {
      console.log(`Received ${messageType} for dialog ${dialogId}:`, chunk)
    }
  })

  const handleJoinDialog = (dialogId: string) => {
    subscribeToDialog(dialogId)
  }

  return (
    <div>
      <div>Connection: {connectionState}</div>
      {Array.from(subscribedDialogs).map(dialogId => (
        <DialogSubscription
          key={dialogId}
          dialogId={dialogId}
          isActive={dialogId === 'dialog-123'}
          processChunk={(targetDialogId, chunk, messageType) => {
            // Process incoming message chunks
          }}
        />
      ))}
    </div>
  )
}
```
