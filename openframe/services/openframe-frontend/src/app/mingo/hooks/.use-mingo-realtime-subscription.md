A React hook and component for managing realtime NATS subscriptions for Mingo chat dialogs with catchup functionality and multi-topic support.

## Key Components

- **`useMingoRealtimeSubscription`** - Main hook for managing multiple dialog subscriptions with state tracking
- **`DialogSubscription`** - Component that handles individual dialog NATS connections and message processing
- **`getApiBaseUrl()`** - Utility to resolve the API base URL from runtime config or window location
- **`getAccessToken()`** - Helper to retrieve access token from localStorage for authentication

## Usage Example

```typescript
// In a chat component
function ChatContainer() {
  const [activeDialogId, setActiveDialogId] = useState<string | null>(null)
  
  const { subscribeToDialog, subscribedDialogs, connectionState } = 
    useMingoRealtimeSubscription(activeDialogId, {
      onChunkReceived: (dialogId, chunk, messageType) => {
        console.log('Received chunk for dialog:', dialogId, chunk)
      }
    })
  
  // Subscribe to a dialog
  const handleOpenDialog = (dialogId: string) => {
    setActiveDialogId(dialogId)
    subscribeToDialog(dialogId)
  }
  
  // Render subscription components for each dialog
  return (
    <div>
      {Array.from(subscribedDialogs).map(dialogId => (
        <DialogSubscription
          key={dialogId}
          dialogId={dialogId}
          isActive={dialogId === activeDialogId}
          processChunk={(targetDialogId, chunk, messageType) => {
            // Handle incoming message chunks
            updateMessageStore(targetDialogId, chunk)
          }}
        />
      ))}
    </div>
  )
}
```
