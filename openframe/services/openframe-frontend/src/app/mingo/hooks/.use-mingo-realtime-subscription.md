A React hook for managing real-time NATS subscriptions to Mingo chat dialogs, providing unified subscription management, chunk processing, and message state synchronization across multiple chat conversations.

## Key Components

- **useMingoRealtimeSubscription** - Main hook for managing multiple dialog subscriptions with connection state tracking
- **useDialogChunkProcessor** - Per-dialog chunk processing with streaming message handling and approval workflows
- **DialogSubscription** - Component wrapper for individual dialog subscriptions with catch-up functionality
- **getApiBaseUrl/getAccessToken** - Utility functions for runtime configuration and authentication
- **MINGO_TOPICS** - Supported NATS message types for admin messages

## Usage Example

```typescript
import { useMingoRealtimeSubscription, DialogSubscription } from './use-mingo-realtime-subscription'

// Main subscription manager
function ChatManager() {
  const {
    subscribeToDialog,
    unsubscribeFromDialog,
    getSubscriptionState,
    subscribedDialogs,
    connectionState
  } = useMingoRealtimeSubscription(activeDialogId, {
    onChunkReceived: (dialogId, chunk, messageType) => {
      console.log('Chunk received for dialog:', dialogId)
    }
  })

  const handleJoinDialog = (dialogId: string) => {
    subscribeToDialog(dialogId)
  }

  return (
    <div>
      <p>Connection: {connectionState}</p>
      {Array.from(subscribedDialogs).map(dialogId => (
        <DialogSubscription
          key={dialogId}
          dialogId={dialogId}
          isActive={dialogId === activeDialogId}
          onApprove={(requestId) => handleApproval(requestId)}
          onReject={(requestId) => handleRejection(requestId)}
          approvalStatuses={approvalStatuses}
        />
      ))}
    </div>
  )
}
```
