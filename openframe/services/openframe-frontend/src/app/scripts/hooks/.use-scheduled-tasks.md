React hook for managing scheduled tasks in Tactical RMM, providing CRUD operations with React Query integration for caching and real-time updates.

## Key Components

- **`ScheduledTask`** - Main interface defining task structure with scheduling configuration, actions, and metadata
- **`ScheduledTaskAction`** - Interface for task actions, currently supporting script execution with parameters
- **`useScheduledTasks`** - Primary hook for fetching and managing scheduled tasks by script ID
- **`scheduledTasksQueryKeys`** - Query key factory for consistent cache management
- **API functions** - Async functions for fetching and deleting scheduled tasks via Tactical API client

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScheduledTasksList({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => {
        console.log('Task deleted successfully')
      },
      onError: (error) => {
        console.error('Failed to delete task:', error.message)
      }
    })
  }

  if (isLoading) return <div>Loading tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <p>Enabled: {task.enabled ? 'Yes' : 'No'}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
