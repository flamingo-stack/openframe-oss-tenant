A React hook for managing scheduled tasks in Tactical RMM, providing functionality to fetch tasks by script ID and delete individual tasks with automatic cache invalidation.

## Key Components

- **`ScheduledTask` Interface**: Comprehensive type definition for scheduled tasks with support for daily/weekly/monthly/runonce scheduling patterns
- **`ScheduledTaskAction` Interface**: Type for task actions, currently supporting script execution with configurable parameters
- **`useScheduledTasks` Hook**: Main hook providing query state management and mutation functions for scheduled task operations
- **Query Key Factory**: Structured query keys for efficient caching and invalidation
- **API Functions**: Wrapper functions for Tactical RMM API calls with error handling

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => {
        console.log('Task deleted successfully')
      },
      onError: (error) => {
        console.error('Failed to delete task:', error.message)
      }
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <p>Enabled: {task.enabled ? 'Yes' : 'No'}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
