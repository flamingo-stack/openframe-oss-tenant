A React hook for managing scheduled tasks associated with scripts, providing CRUD operations and filtering capabilities through React Query.

## Key Components

- **`ScheduledTask` Interface** - Complete type definition for scheduled tasks with timing, actions, and configuration
- **`ScheduledTaskAction` Interface** - Type for script-based actions within scheduled tasks
- **`useScheduledTasks` Hook** - Main hook for querying and mutating scheduled tasks filtered by script ID
- **`scheduledTasksQueryKeys`** - Query key factory for consistent caching and invalidation
- **API Functions** - `fetchScheduledTasksByScript()` and `deleteScheduledTaskApi()` for server communication

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error)
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
