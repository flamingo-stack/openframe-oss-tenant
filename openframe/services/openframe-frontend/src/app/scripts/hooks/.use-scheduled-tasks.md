A React hook that manages scheduled tasks data from the Tactical RMM API, providing CRUD operations and real-time state management using TanStack Query.

## Key Components

- **`ScheduledTask` interface** - Complete type definition for scheduled tasks with timing, actions, and configuration
- **`ScheduledTaskAction` interface** - Type for script-based task actions with execution parameters
- **`useScheduledTasks` hook** - Main hook for fetching and managing scheduled tasks filtered by script ID
- **`scheduledTasksQueryKeys`** - Centralized query key factory for consistent caching
- **API functions** - `fetchScheduledTasksByScript` and `deleteScheduledTaskApi` for server communication

## Usage Example

```typescript
import { useScheduledTasks } from '@/hooks/use-scheduled-tasks'

function ScheduledTasksManager({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDelete = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error)
    })
  }

  if (isLoading) return <div>Loading tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <button 
            onClick={() => handleDelete(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
