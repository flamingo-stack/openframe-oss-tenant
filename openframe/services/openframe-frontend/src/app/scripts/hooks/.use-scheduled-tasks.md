A React hook for managing scheduled tasks in the Tactical RMM system, providing query and mutation operations for script-related scheduled tasks.

## Key Components

- **`ScheduledTask`** - TypeScript interface defining the complete structure of a scheduled task including timing, actions, and configuration
- **`ScheduledTaskAction`** - Interface for task actions that can execute scripts with parameters and environment variables  
- **`useScheduledTasks(scriptId)`** - Main hook that fetches scheduled tasks associated with a specific script ID
- **`scheduledTasksQueryKeys`** - Query key factory for React Query cache management
- **`deleteTask`** - Mutation function for removing scheduled tasks with success/error callbacks

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => {
        console.log('Task deleted successfully')
      },
      onError: (error) => {
        console.error('Failed to delete task:', error.message)
      }
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <p>Actions: {task.actions.length}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
