A React Query hook for managing scheduled tasks that are associated with specific scripts in the Tactical RMM system.

## Key Components

- **`ScheduledTask`** - Interface defining the complete structure of a scheduled task with timing, actions, and configuration options
- **`ScheduledTaskAction`** - Interface for script-based actions that can be executed as part of a scheduled task
- **`useScheduledTasks(scriptId)`** - Main hook that provides query and mutation functionality for script-specific scheduled tasks
- **`scheduledTasksQueryKeys`** - Query key factory for consistent caching and invalidation
- **`fetchScheduledTasksByScript()`** - API function that retrieves and filters scheduled tasks by script ID
- **`deleteScheduledTaskApi()`** - API function for removing scheduled tasks

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => {
        console.log('Task deleted successfully')
      },
      onError: (error) => {
        console.error('Failed to delete task:', error.message)
      }
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <p>Enabled: {task.enabled ? 'Yes' : 'No'}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
