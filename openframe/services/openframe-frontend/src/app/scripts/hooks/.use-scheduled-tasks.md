A React hook for managing scheduled tasks with TanStack Query integration, providing CRUD operations and real-time state management for tasks filtered by script ID.

## Key Components

- **`ScheduledTask`** - Complete interface for scheduled task data with timing, actions, and configuration
- **`ScheduledTaskAction`** - Interface for script-based actions within tasks
- **`scheduledTasksQueryKeys`** - Query key factory for cache management
- **`useScheduledTasks`** - Main hook providing filtered tasks, loading states, and delete operations
- **`fetchScheduledTasksByScript`** - API function to retrieve tasks filtered by script ID
- **`deleteScheduledTaskApi`** - API function to remove scheduled tasks

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScheduledTasksManager({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => {
        console.log('Task deleted successfully')
      },
      onError: (error) => {
        console.error('Delete failed:', error.message)
      }
    })
  }

  if (isLoading) return <div>Loading tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
