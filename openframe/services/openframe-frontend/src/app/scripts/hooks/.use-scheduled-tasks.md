A React hook for managing scheduled tasks in Tactical RMM, providing functionality to fetch tasks by script ID and delete them with React Query integration.

## Key Components

**Types:**
- `ScheduledTaskAction` - Defines script-based task actions with execution parameters
- `ScheduledTask` - Complete task configuration including timing, recurrence, and actions

**Hook:**
- `useScheduledTasks` - Main hook for fetching and managing scheduled tasks
- `scheduledTasksQueryKeys` - Query key factory for cache management

**API Functions:**
- `fetchScheduledTasksByScript` - Retrieves tasks filtered by script ID
- `deleteScheduledTaskApi` - Removes a scheduled task

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScheduledTasksList({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDelete = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Failed to delete:', error.message)
    })
  }

  if (isLoading) return <div>Loading tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <button 
            onClick={() => handleDelete(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
      <button onClick={() => refetch()}>Refresh</button>
    </div>
  )
}
```
