A React Query hook for managing scheduled tasks in the Tactical RMM system, providing CRUD operations and filtering capabilities for script-related tasks.

## Key Components

- **`ScheduledTask`** - Interface defining task properties including timing, actions, and execution settings
- **`ScheduledTaskAction`** - Interface for task actions that execute scripts with configurable parameters
- **`useScheduledTasks`** - Main hook providing query and mutation operations for scheduled tasks
- **`scheduledTasksQueryKeys`** - Query key factory for consistent cache management
- **`fetchScheduledTasksByScript`** - API function to retrieve tasks filtered by script ID
- **`deleteScheduledTaskApi`** - API function to remove scheduled tasks

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScheduledTasksManager({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => {
        console.log('Task deleted successfully')
      },
      onError: (error) => {
        console.error('Failed to delete task:', error.message)
      }
    })
  }

  if (isLoading) return <div>Loading tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
