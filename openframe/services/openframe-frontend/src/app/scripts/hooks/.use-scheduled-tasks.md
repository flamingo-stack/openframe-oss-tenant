Custom React hook for managing scheduled tasks filtered by script ID using TanStack Query for data fetching and caching. Provides filtered task data, loading states, and delete functionality with automatic cache invalidation.

## Key Components

**Types:**
- `ScheduledTaskAction` - Interface for task actions (script execution with timeout, args, env vars)
- `ScheduledTask` - Complete task interface with scheduling options (daily/weekly/monthly/runonce)

**Query Keys:**
- `scheduledTasksQueryKeys` - Hierarchical query key factory for cache management

**Main Hook:**
- `useScheduledTasks(scriptId)` - Returns filtered tasks, loading states, and delete mutation

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDelete = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error)
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      <h3>Scheduled Tasks ({scheduledTasks.length})</h3>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <span>{task.name} - {task.task_type}</span>
          <button 
            onClick={() => handleDelete(task.id.toString())}
            disabled={isDeleting}
          >
            Delete
          </button>
        </div>
      ))}
    </div>
  )
}
```
