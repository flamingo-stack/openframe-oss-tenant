Custom React hook for managing scheduled tasks in the Tactical RMM integration, providing query and mutation operations for tasks associated with specific scripts.

## Key Components

- **`ScheduledTask` Interface**: Defines the complete structure for scheduled tasks including timing, actions, and configuration
- **`ScheduledTaskAction` Interface**: Represents individual actions within a task (currently supports script execution)
- **`useScheduledTasks` Hook**: Main hook providing CRUD operations for scheduled tasks
- **`scheduledTasksQueryKeys`**: Query key factory for proper cache management
- **API Functions**: `fetchScheduledTasksByScript` and `deleteScheduledTaskApi` for backend communication

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error.message)
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
