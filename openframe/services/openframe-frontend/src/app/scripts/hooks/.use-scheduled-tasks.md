A React Query hook for managing scheduled tasks filtered by script ID, providing data fetching, deletion capabilities, and loading states.

## Key Components

- **`ScheduledTask`** - Interface defining the structure of scheduled tasks with timing configurations and actions
- **`ScheduledTaskAction`** - Interface for script-based actions within scheduled tasks
- **`scheduledTasksQueryKeys`** - Query key factory for React Query caching strategies
- **`useScheduledTasks`** - Main hook providing scheduled tasks management for a specific script
- **`fetchScheduledTasksByScript`** - API function to retrieve and filter tasks by script ID
- **`deleteScheduledTaskApi`** - API function to delete a scheduled task

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error.message)
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
