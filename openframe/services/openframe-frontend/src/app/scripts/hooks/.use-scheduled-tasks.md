A React hook for managing scheduled tasks filtered by script ID, providing query and mutation operations through TanStack Query.

## Key Components

- **ScheduledTask/ScheduledTaskAction interfaces**: TypeScript definitions for task data structures supporting daily/weekly/monthly scheduling with script actions
- **scheduledTasksQueryKeys**: Centralized query key factory for cache management
- **fetchScheduledTasksByScript()**: API function that retrieves and filters scheduled tasks by script ID
- **deleteScheduledTaskApi()**: API function for task deletion
- **useScheduledTasks()**: Main hook providing filtered tasks, loading states, and delete functionality

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptTasksPanel({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error)
    })
  }

  if (isLoading) return <div>Loading tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
