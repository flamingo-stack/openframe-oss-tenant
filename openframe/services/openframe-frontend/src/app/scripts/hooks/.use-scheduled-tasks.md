A React Query hook for managing scheduled tasks from the Tactical RMM API, providing CRUD operations with loading states and optimistic updates.

## Key Components

**Types:**
- `ScheduledTaskAction` - Defines script execution configuration with timeout and environment variables
- `ScheduledTask` - Complete task model with scheduling options (daily/weekly/monthly/runonce) and execution settings

**Query Keys:**
- `scheduledTasksQueryKeys` - Structured cache keys for React Query invalidation and deduplication

**Hook Functions:**
- `useScheduledTasks()` - Main hook providing scheduled tasks data and operations
- `fetchScheduledTasksByScript()` - API wrapper for retrieving tasks filtered by script ID
- `deleteScheduledTaskApi()` - API wrapper for task deletion

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const { 
    scheduledTasks, 
    isLoading, 
    error, 
    deleteTask, 
    isDeleting 
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error.message)
    })
  }

  if (isLoading) return <div>Loading tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <span>{task.name} ({task.task_type})</span>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete
          </button>
        </div>
      ))}
    </div>
  )
}
```
