A React hook for managing scheduled tasks associated with scripts, providing query and mutation functionality for the TacticalRMM platform.

## Key Components

- **`ScheduledTask`** - Interface defining task structure with scheduling options (daily, weekly, monthly, runonce)
- **`ScheduledTaskAction`** - Interface for script actions within tasks
- **`scheduledTasksQueryKeys`** - Query key factory for React Query caching
- **`useScheduledTasks`** - Main hook providing task data and operations for a specific script
- **`fetchScheduledTasksByScript`** - API function to retrieve tasks filtered by script ID
- **`deleteScheduledTaskApi`** - API function to remove scheduled tasks

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Failed to delete task:', error)
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      <h3>Scheduled Tasks ({scheduledTasks.length})</h3>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <span>{task.name} - {task.task_type}</span>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete
          </button>
        </div>
      ))}
    </div>
  )
}
```
