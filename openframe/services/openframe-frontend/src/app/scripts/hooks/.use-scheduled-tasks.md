A React Query hook for managing scheduled tasks in TacticalRMM, providing functionality to fetch tasks by script ID and delete individual tasks with optimistic cache updates.

## Key Components

- **`ScheduledTask`** - Complete task interface with scheduling options (daily/weekly/monthly/runonce)
- **`ScheduledTaskAction`** - Task action definition for script execution with parameters
- **`useScheduledTasks`** - Main hook for fetching and managing scheduled tasks
- **`scheduledTasksQueryKeys`** - Centralized query key factory for cache management
- **`deleteTask`** - Mutation function with success/error callback support

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error.message)
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
