React Query hook for managing scheduled tasks from Tactical RMM, providing filtering by script ID and task deletion capabilities with optimistic updates.

## Key Components

- **`ScheduledTask`** - Complete interface for scheduled task data including timing configuration, actions, and metadata
- **`ScheduledTaskAction`** - Interface for script-based actions within scheduled tasks
- **`useScheduledTasks(scriptId)`** - Main hook for fetching and managing scheduled tasks filtered by script ID
- **`scheduledTasksQueryKeys`** - Query key factory for consistent cache management
- **`deleteTask()`** - Mutation function with success/error callbacks

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDelete = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Failed to delete:', error.message)
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <p>Enabled: {task.enabled ? 'Yes' : 'No'}</p>
          <button 
            onClick={() => handleDelete(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
      <button onClick={() => refetch()}>Refresh Tasks</button>
    </div>
  )
}
```
