A React Query hook for managing scheduled tasks that execute scripts, providing CRUD operations and filtering functionality for task scheduling workflows.

## Key Components

- **`ScheduledTask`** - Interface defining scheduled task structure with timing, actions, and configuration options
- **`ScheduledTaskAction`** - Interface for script execution actions within scheduled tasks
- **`scheduledTasksQueryKeys`** - Query key factory for consistent cache management
- **`useScheduledTasks(scriptId)`** - Main hook providing scheduled task operations filtered by script ID
- **`fetchScheduledTasksByScript()`** - API function to retrieve and filter scheduled tasks by script
- **`deleteScheduledTaskApi()`** - API function for task deletion

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduleManager({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => {
        console.log('Task deleted successfully')
      },
      onError: (error) => {
        console.error('Delete failed:', error.message)
      }
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <p>Enabled: {task.enabled ? 'Yes' : 'No'}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
