React Query hook for managing scheduled tasks in the Tactical RMM platform, providing data fetching, filtering by script ID, and deletion functionality.

## Key Components

- **`ScheduledTask`** - TypeScript interface defining task structure with scheduling configuration, actions, and metadata
- **`ScheduledTaskAction`** - Interface for script-based task actions with execution parameters
- **`useScheduledTasks(scriptId)`** - Main hook for fetching and managing scheduled tasks filtered by script
- **`scheduledTasksQueryKeys`** - Query key factory for React Query caching
- **`deleteTask`** - Mutation function for removing scheduled tasks with callback support

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptTasksView({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error.message)
    })
  }

  if (isLoading) return <div>Loading tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
