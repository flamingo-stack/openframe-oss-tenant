A React Query hook for managing scheduled tasks, providing operations to fetch tasks filtered by script ID and delete individual tasks.

## Key Components

### Types
- **`ScheduledTask`** - Complete task definition including scheduling parameters, actions, and metadata
- **`ScheduledTaskAction`** - Action configuration for script execution with timeout and environment settings
- **`scheduledTasksQueryKeys`** - Query key factory for consistent cache management

### Hook Export
- **`useScheduledTasks(scriptId: string)`** - Main hook providing scheduled task operations for a specific script

### API Functions
- **`fetchScheduledTasksByScript(scriptId: string)`** - Fetches and filters tasks containing actions for the specified script
- **`deleteScheduledTaskApi(taskId: string)`** - Removes a scheduled task via the Tactical RMM API

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScheduledTasksManager({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => {
        console.log('Task deleted successfully')
      },
      onError: (error) => {
        console.error('Failed to delete task:', error.message)
      }
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <span>{task.name} - {task.task_type}</span>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete
          </button>
        </div>
      ))}
    </div>
  )
}
```
