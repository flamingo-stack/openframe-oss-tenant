A React Query hook for managing scheduled tasks in the Tactical RMM system, providing CRUD operations with automatic cache management and optimistic updates.

## Key Components

- **`ScheduledTask`** - Main interface defining task properties including schedule type, timing, actions, and configuration
- **`ScheduledTaskAction`** - Interface for script actions within scheduled tasks with execution parameters
- **`scheduledTasksQueryKeys`** - Standardized query key factory for cache management
- **`useScheduledTasks`** - Primary hook for fetching and managing scheduled tasks filtered by script ID
- **`fetchScheduledTasksByScript`** - API function to retrieve tasks associated with a specific script
- **`deleteScheduledTaskApi`** - API function for task deletion with error handling

## Usage Example

```typescript
import { useScheduledTasks } from './use-scheduled-tasks'

function ScriptScheduledTasks({ scriptId }: { scriptId: string }) {
  const {
    scheduledTasks,
    isLoading,
    error,
    deleteTask,
    isDeleting,
    refetch
  } = useScheduledTasks(scriptId)

  const handleDeleteTask = (taskId: string) => {
    deleteTask(taskId, {
      onSuccess: () => console.log('Task deleted successfully'),
      onError: (error) => console.error('Delete failed:', error.message)
    })
  }

  if (isLoading) return <div>Loading scheduled tasks...</div>
  if (error) return <div>Error: {error}</div>

  return (
    <div>
      {scheduledTasks.map(task => (
        <div key={task.id}>
          <h3>{task.name}</h3>
          <p>Type: {task.task_type}</p>
          <p>Enabled: {task.enabled ? 'Yes' : 'No'}</p>
          <button 
            onClick={() => handleDeleteTask(task.id.toString())}
            disabled={isDeleting}
          >
            Delete Task
          </button>
        </div>
      ))}
    </div>
  )
}
```
