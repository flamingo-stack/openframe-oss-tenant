Defines TypeScript interfaces for osquery query management, including query metadata, report results, and query report request parameters.

## Key Components

- **`Query`** — Full query entity with metadata (author, scheduling, platform targeting, automations) and execution state fields
- **`QueryReportResult`** — Single host result row from a query report, containing host identity and a dynamic column map
- **`QueryReportResponse`** — API response wrapper for query reports, including clipping status and result set
- **`QueryReportParams`** — Optional sort parameters for fetching query reports

## Usage Example

```typescript
import {
  Query,
  QueryReportResponse,
  QueryReportParams,
} from "./queries.types";

const query: Query = {
  id: 42,
  name: "Running Processes",
  query: "SELECT * FROM processes;",
  description: "Lists all running processes",
  author_id: 1,
  author_name: "Alice",
  author_email: "alice@example.com",
  saved: true,
  observer_can_run: false,
  interval: 3600,
  automations_enabled: true,
  created_at: "2024-01-01T00:00:00Z",
  updated_at: "2024-06-01T00:00:00Z",
};

const params: QueryReportParams = {
  order_key: "last_fetched",
  order_direction: "desc",
};

const report: QueryReportResponse = {
  query_id: 42,
  report_clipped: false,
  results: [
    {
      host_id: 7,
      host_name: "macbook-alice",
      last_fetched: "2024-06-01T12:00:00Z",
      columns: { pid: "1234", name: "node" },
    },
  ],
};
```
