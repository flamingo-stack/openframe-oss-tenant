Defines standardized query keys for React Query operations related to tickets/dialogs, providing type-safe caching and invalidation patterns for paginated dialog data with filtering capabilities.

## Key Components

- **`DialogsQueryParams`** - Interface defining parameters for dialog queries (archived status, search, filters, pagination cursor)
- **`dialogsQueryKeys`** - Hierarchical query key factory with base keys, list queries, and parameterized list queries
- **`invalidateAllDialogs`** - Utility function to invalidate all dialog-related queries in React Query cache

## Usage Example

```typescript
import { useQuery, useQueryClient } from '@tanstack/react-query'
import { dialogsQueryKeys, invalidateAllDialogs, DialogsQueryParams } from './query-keys'

// Using query keys in a React Query hook
const useDialogs = (params: DialogsQueryParams) => {
  return useQuery({
    queryKey: dialogsQueryKeys.list(params),
    queryFn: () => fetchDialogs(params)
  })
}

// Invalidating queries after mutation
const queryClient = useQueryClient()
const handleDialogUpdate = async () => {
  // Perform update...
  await invalidateAllDialogs(queryClient)
}

// Query with specific parameters
const dialogParams: DialogsQueryParams = {
  archived: false,
  search: 'urgent',
  statusFilters: ['open', 'pending'],
  cursor: 'page_2'
}
```
