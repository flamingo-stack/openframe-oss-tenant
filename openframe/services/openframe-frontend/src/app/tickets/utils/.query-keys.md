A type-safe query key factory for managing React Query cache keys related to tickets/dialogs, providing consistent cache invalidation and hierarchical key structure.

## Key Components

- **`DialogsQueryParams`** - Interface defining filter parameters for dialog queries (archived status, search, status filters, cursor pagination)
- **`dialogsQueryKeys`** - Hierarchical query key factory with base keys (`all`, `lists`) and parameterized keys (`list`)
- **`invalidateAllDialogs`** - Utility function to invalidate all dialog-related queries in the React Query cache

## Usage Example

```typescript
import { useQuery, useQueryClient } from '@tanstack/react-query'
import { dialogsQueryKeys, invalidateAllDialogs, DialogsQueryParams } from './query-keys'

// Using query keys in a hook
const useDialogs = (params: DialogsQueryParams) => {
  return useQuery({
    queryKey: dialogsQueryKeys.list(params),
    queryFn: () => fetchDialogs(params)
  })
}

// Invalidating queries after mutation
const queryClient = useQueryClient()
const handleDialogUpdate = async () => {
  await updateDialog(dialogId)
  // Invalidate all dialog queries
  await invalidateAllDialogs(queryClient)
}

// Example query parameters
const queryParams: DialogsQueryParams = {
  archived: false,
  search: 'urgent',
  statusFilters: ['open', 'in-progress'],
  cursor: 'page-2'
}
```
