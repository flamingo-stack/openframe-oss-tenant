Defines structured query keys for React Query cache management in the dialogs/tickets system, providing type-safe query invalidation and parameter handling.

## Key Components

- **`DialogsQueryParams`** - Interface defining query parameters including archived status, search terms, status filters, and pagination cursor
- **`dialogsQueryKeys`** - Hierarchical query key factory with base keys, list queries, and parameterized queries
- **`invalidateAllDialogs`** - Utility function to invalidate all dialog-related queries in the cache

## Usage Example

```typescript
import { useQuery, useQueryClient } from '@tanstack/react-query'
import { dialogsQueryKeys, invalidateAllDialogs } from './query-keys'

// Using structured query keys in a hook
const useDialogs = (params: DialogsQueryParams) => {
  return useQuery({
    queryKey: dialogsQueryKeys.list(params),
    queryFn: () => fetchDialogs(params)
  })
}

// Invalidating queries after mutation
const queryClient = useQueryClient()
await createDialog(newDialog)
invalidateAllDialogs(queryClient)

// Accessing different query key levels
const allDialogsKey = dialogsQueryKeys.all        // ['dialogs']
const listsKey = dialogsQueryKeys.lists()         // ['dialogs', 'list']
const specificKey = dialogsQueryKeys.list({       // ['dialogs', 'list', {...params}]
  archived: false,
  search: 'urgent'
})
```
