A React hook that processes realtime NATS messages from AI dialog streams, converting raw payloads into structured Message objects and triggering appropriate UI updates for both client and admin chat contexts.

## Key Components

- **`useDialogRealtimeProcessor`** - Main hook that returns a `processChunk` function for handling realtime messages
- **`processChunk`** - Core processor that parses NATS payloads, determines message types, and creates Message objects
- **Message Type Handling** - Supports text messages, tool executions, approval requests/results, and error messages
- **Admin/Client Context** - Differentiates between admin and client chat streams with appropriate message attribution

## Usage Example

```typescript
import { useDialogRealtimeProcessor } from './use-dialog-realtime-processor'

function DialogComponent({ dialogId }: { dialogId: string }) {
  const [isStreaming, setIsStreaming] = useState(false)
  const [messages, setMessages] = useState<Message[]>([])

  const { processChunk } = useDialogRealtimeProcessor({
    dialogId,
    onStreamStart: (isAdmin) => setIsStreaming(true),
    onStreamEnd: (isAdmin) => setIsStreaming(false),
    onMessageAdd: (message, isAdmin) => {
      setMessages(prev => [...prev, message])
    },
    onError: (error, isAdmin) => {
      console.error('Dialog error:', error)
    }
  })

  // Connect to NATS and process incoming messages
  useEffect(() => {
    const subscription = natsClient.subscribe(`dialog.${dialogId}`)
    subscription.callback = (msg) => processChunk(msg.data, 'message')
    
    return () => subscription.unsubscribe()
  }, [dialogId, processChunk])

  return (
    <div>
      {messages.map(msg => <MessageComponent key={msg.id} message={msg} />)}
      {isStreaming && <StreamingIndicator />}
    </div>
  )
}
```
