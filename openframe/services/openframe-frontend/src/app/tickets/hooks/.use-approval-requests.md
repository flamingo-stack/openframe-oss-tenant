A React hook for managing approval request actions using TanStack Query mutations to approve or reject requests with proper error handling and loading states.

## Key Components

- **`ApprovalRequestAction`** - Type definition for approval request parameters (requestId and approve boolean)
- **`approveRequest`** - Core API function that sends approval/rejection requests to the backend
- **`useApprovalRequests`** - Main React hook that provides approval workflow functionality
- **`handleApproveRequest`** - Helper method to approve a request with success/error callbacks
- **`handleRejectRequest`** - Helper method to reject a request with success/error callbacks

## Usage Example

```typescript
import { useApprovalRequests } from './use-approval-requests'

function ApprovalComponent() {
  const {
    handleApproveRequest,
    handleRejectRequest,
    isLoading,
    error
  } = useApprovalRequests()

  const onApprove = async (requestId: string) => {
    await handleApproveRequest(requestId, {
      onSuccess: (status) => console.log('Approved:', status),
      onError: (error) => console.error('Failed:', error)
    })
  }

  const onReject = async (requestId: string) => {
    await handleRejectRequest(requestId, {
      onSuccess: (status) => console.log('Rejected:', status),
      onError: (error) => console.error('Failed:', error)
    })
  }

  return (
    <div>
      <button 
        onClick={() => onApprove('req-123')}
        disabled={isLoading}
      >
        {isLoading ? 'Processing...' : 'Approve'}
      </button>
      <button 
        onClick={() => onReject('req-123')}
        disabled={isLoading}
      >
        Reject
      </button>
      {error && <div className="error">{error}</div>}
    </div>
  )
}
```
