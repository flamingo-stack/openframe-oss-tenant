A React hook for managing approval request actions (approve/reject) using React Query mutations with error handling and status tracking.

## Key Components

- **`ApprovalRequestAction`** - Type defining the structure for approval actions with request ID and approval boolean
- **`approveRequest()`** - API function that sends approval/rejection requests to the backend endpoint
- **`useApprovalRequests()`** - Main hook providing approval and rejection handlers with loading states
- **`handleApproveRequest()` / `handleRejectRequest()`** - Convenience methods for approving/rejecting requests with optional callbacks

## Usage Example

```typescript
import { useApprovalRequests } from './use-approval-requests'

function ApprovalComponent() {
  const { 
    handleApproveRequest, 
    handleRejectRequest, 
    isLoading, 
    error 
  } = useApprovalRequests()

  const onApprove = async (requestId: string) => {
    await handleApproveRequest(requestId, {
      onSuccess: (status) => console.log('Approved:', status),
      onError: (error) => console.error('Failed:', error)
    })
  }

  const onReject = async (requestId: string) => {
    await handleRejectRequest(requestId, {
      onSuccess: (status) => console.log('Rejected:', status),
      onError: (error) => console.error('Failed:', error)
    })
  }

  return (
    <div>
      <button 
        onClick={() => onApprove('req-123')} 
        disabled={isLoading}
      >
        Approve
      </button>
      <button 
        onClick={() => onReject('req-123')} 
        disabled={isLoading}
      >
        Reject
      </button>
      {error && <p>Error: {error}</p>}
    </div>
  )
}
```
