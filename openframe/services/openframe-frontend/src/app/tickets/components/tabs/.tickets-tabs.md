A Next.js client-side component that provides tabbed navigation for tickets/chat management, with current and archived chat views.

## Key Components

- **`TICKETS_TABS`** - Configuration array defining the two tab items (current and archived chats)
- **`getTicketsTab()`** - Utility function to find a tab by ID
- **`getTabComponent()`** - Utility function to retrieve the React component for a specific tab
- **`TicketsTabNavigation`** - Main navigation component that handles tab switching and URL management

## Usage Example

```typescript
import { TicketsTabNavigation, getTabComponent } from './tickets-tabs'

function TicketsPage() {
  const [activeTab, setActiveTab] = useState('current')

  // Basic usage with default URL handling
  return (
    <div>
      <TicketsTabNavigation activeTab={activeTab} />
    </div>
  )
}

// Custom tab change handler
function CustomTicketsPage() {
  const handleTabChange = (tabId: string) => {
    console.log('Switching to tab:', tabId)
    // Custom logic here
  }

  return (
    <TicketsTabNavigation 
      activeTab="archived"
      onTabChange={handleTabChange} 
    />
  )
}

// Get specific tab component dynamically
const TabComponent = getTabComponent('current') // Returns CurrentChats
```

The component automatically clears URL parameters when switching tabs to provide a clean slate for each view, and integrates with Next.js routing for URL synchronization.
